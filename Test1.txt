"""
Abstract base class for extractors.
"""

from __future__ import annotations

from abc import ABC, abstractmethod
from typing import Iterator

import pandas as pd


class BaseExtractor(ABC):
    """Extract data from a source and yield DataFrames."""

    @abstractmethod
    def extract(self, chunk_size: int | None = None) -> Iterator[pd.DataFrame]:
        """Yield DataFrames, optionally in chunks."""
        ...

    @abstractmethod
    def extract_all(self) -> pd.DataFrame:
        """Extract the entire dataset into a single DataFrame."""
        ...
