"""
Truncate and load loader for databases.
"""

from __future__ import annotations

import pandas as pd
from sqlalchemy import text
from sqlalchemy.engine import Engine

from loaders.base import BaseLoader
from utils.logger import get_logger

logger = get_logger("truncate_loader")


class TruncateLoader(BaseLoader):
    """Truncate the target table, then load data."""

    def __init__(self, engine: Engine, table: str, schema: str | None = None, batch_size: int = 10000):
        self.engine = engine
        self.table = table
        self.schema = schema
        self.batch_size = batch_size
        self._truncated = False

    def load(self, df: pd.DataFrame) -> dict:
        if df.empty:
            return {"rows_loaded": 0}

        if not self._truncated:
            table_name = f"{self.schema}.{self.table}" if self.schema else self.table
            try:
                with self.engine.begin() as conn:
                    # SQLite doesn't support TRUNCATE, so delete
                    if self.engine.dialect.name == "sqlite":
                        conn.execute(text(f"DELETE FROM {table_name}"))
                    else:
                        conn.execute(text(f"TRUNCATE TABLE {table_name}"))
                logger.info("Truncated table %s", table_name)
            except Exception as exc:
                logger.warning("Truncate failed, attempting drop/create or delete: %s", exc)
                # Fallback to delete
                try:
                    with self.engine.begin() as conn:
                        conn.execute(text(f"DELETE FROM {table_name}"))
                except Exception as e2:
                    logger.error("Fallback delete also failed: %s", e2)
                    
            self._truncated = True

        logger.info("Loading %d rows into %s (batch_size=%d)", len(df), self.table, self.batch_size)
        try:
            df.to_sql(
                name=self.table,
                con=self.engine,
                schema=self.schema,
                if_exists="append",
                index=False,
                chunksize=self.batch_size,
                method="multi"  # multi-row insert
            )
            return {"rows_loaded": len(df), "rows_inserted": len(df)}
        except Exception as exc:
            logger.error("TruncateLoad insert failed: %s", exc)
            raise
