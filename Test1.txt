"""
Oracle database connector using cx_Oracle + SQLAlchemy.
"""

from __future__ import annotations

import os
from typing import Any

from connectors.base import BaseConnector
from utils.logger import get_logger

logger = get_logger("oracle_connector")


class OracleConnector(BaseConnector):
    """Connector for Oracle databases."""

    def __init__(self, **kwargs: Any):
        super().__init__(**kwargs)
        self.host = kwargs.get("host") or os.getenv("ORACLE_HOST", "localhost")
        self.port = kwargs.get("port") or int(os.getenv("ORACLE_PORT", "1521"))
        self.service_name = kwargs.get("service_name") or os.getenv("ORACLE_SERVICE", "")
        self.user = kwargs.get("user") or os.getenv("ORACLE_USER", "")
        self.password = kwargs.get("password") or os.getenv("ORACLE_PASSWORD", "")
        self.connection_url = kwargs.get("connection_url")

    def build_connection_url(self) -> str:
        if self.connection_url:
            return self.connection_url

        try:
            import cx_Oracle  # noqa: F401
        except ImportError:
            raise ImportError(
                "cx_Oracle is not installed. Install it with: pip install cx_Oracle"
            )

        dsn = f"{self.host}:{self.port}/{self.service_name}"
        url = f"oracle+cx_oracle://{self.user}:{self.password}@{dsn}"
        logger.debug("Oracle URL built for host=%s service=%s", self.host, self.service_name)
        return url
