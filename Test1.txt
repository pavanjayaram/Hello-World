"""
Chunking utilities.
"""

import pandas as pd

def yield_chunks(df: pd.DataFrame, chunk_size: int):
    """Yield chunks from an existing dataframe if native chunking is unavailable."""
    if chunk_size <= 0:
        yield df
        return
        
    for start in range(0, len(df), chunk_size):
        yield df.iloc[start:start+chunk_size]
