"""
Abstract base class for database connectors.
"""

from __future__ import annotations

from abc import ABC, abstractmethod
from typing import Any

from sqlalchemy import create_engine, text
from sqlalchemy.engine import Engine

from utils.logger import get_logger

logger = get_logger("connector")


class BaseConnector(ABC):
    """Abstract connector providing a SQLAlchemy engine."""

    def __init__(self, **kwargs: Any):
        self._engine: Engine | None = None
        self._kwargs = kwargs

    @abstractmethod
    def build_connection_url(self) -> str:
        """Subclass must construct the SQLAlchemy connection URL."""
        ...

    def get_engine(self) -> Engine:
        """Lazy-create and return a SQLAlchemy engine."""
        if self._engine is None:
            url = self.build_connection_url()
            # Resolve .env aliases
            from utils.env_utils import resolve_connection_string
            url = resolve_connection_string(url)
            
            logger.info("Creating engine for: %s", url.split("@")[-1] if "@" in url else url)
            self._engine = create_engine(url, pool_pre_ping=True)
        return self._engine

    def execute(self, sql: str) -> None:
        """Execute a raw SQL statement."""
        engine = self.get_engine()
        with engine.connect() as conn:
            conn.execute(text(sql))
            conn.commit()
        logger.debug("Executed SQL: %s", sql[:200])

    def test_connection(self) -> bool:
        """Test that the connection is alive."""
        try:
            engine = self.get_engine()
            with engine.connect() as conn:
                conn.execute(text("SELECT 1"))
            logger.info("Connection test: OK")
            return True
        except Exception as exc:
            logger.error("Connection test FAILED: %s", exc)
            return False

    def dispose(self) -> None:
        """Dispose of the engine and its connection pool."""
        if self._engine:
            self._engine.dispose()
            self._engine = None
