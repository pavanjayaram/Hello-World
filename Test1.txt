"""
Pandas-based transformer implementing all supported transformation actions.
"""

from __future__ import annotations

from typing import List

import numpy as np
import pandas as pd
import hashlib
import base64

from config.models import TransformationRule
from transformers.base import BaseTransformer
from utils.logger import get_logger

logger = get_logger("pandas_transformer")


class PandasTransformer(BaseTransformer):
    """Apply transformation rules to a pandas DataFrame."""

    def __init__(self):
        super().__init__()
        # State to keep track of row_number/surrogate_key sequences across chunks
        self._sequence_counters = {}
        self._batch_sequence_values = {}
        self.load_id = None
        self.load_timestamp = None

    def set_load_metadata(self, load_id: str, load_timestamp: str):
        """Set execution metadata to be injected into rows."""
        self.load_id = load_id
        self.load_timestamp = load_timestamp

    def set_sequence_start(self, column: str, start_val: int):
        """Pre-seed a sequence counter (e.g. from DB MAX)."""
        self._sequence_counters[column] = int(start_val) if start_val is not None else 0

    def transform(self, df: pd.DataFrame, rules: List[TransformationRule]) -> pd.DataFrame:
        """Apply each rule sequentially and return the transformed DataFrame."""
        result = df.copy()
        for rule in rules:
            action = rule.action.lower()
            handler = getattr(self, f"_action_{action}", None)
            if handler is None:
                logger.warning("Unknown transformation action: '%s' — skipping", action)
                continue
            logger.debug("Applying action='%s' column='%s'", action, rule.column)
            result = handler(result, rule)
        return result

    # ── String transforms ─────────────────────────

    def _action_upper(self, df: pd.DataFrame, rule: TransformationRule) -> pd.DataFrame:
        col = rule.column
        if col and col in df.columns:
            mask = df[col].notna()
            df.loc[mask, col] = df.loc[mask, col].astype(str).str.upper()
        return df

    def _action_lower(self, df: pd.DataFrame, rule: TransformationRule) -> pd.DataFrame:
        col = rule.column
        if col and col in df.columns:
            mask = df[col].notna()
            df.loc[mask, col] = df.loc[mask, col].astype(str).str.lower()
        return df

    def _action_trim(self, df: pd.DataFrame, rule: TransformationRule) -> pd.DataFrame:
        col = rule.column
        if col and col in df.columns:
            mask = df[col].notna()
            df.loc[mask, col] = df.loc[mask, col].astype(str).str.strip()
        return df

    # ── Type conversion ───────────────────────────

    def _action_cast(self, df: pd.DataFrame, rule: TransformationRule) -> pd.DataFrame:
        col = rule.column
        dtype = rule.params.get("dtype", "str")
        fmt = rule.params.get("format")
        formats = rule.params.get("formats")
        
        if col and col in df.columns:
            try:
                if dtype in ("datetime", "date", "timestamp"):
                    if formats and isinstance(formats, list):
                        # Try multiple formats similar to PySpark coalesce(to_timestamp...)
                        success = False
                        for f in formats:
                            try:
                                series = pd.to_datetime(df[col], format=f, errors="raise")
                                df[col] = series
                                success = True
                                break
                            except:
                                continue
                        if not success:
                            df[col] = pd.to_datetime(df[col], errors="coerce")
                    elif fmt:
                        df[col] = pd.to_datetime(df[col], format=fmt, errors="coerce")
                    else:
                        df[col] = pd.to_datetime(df[col], errors="coerce")
                    
                    if dtype == "date":
                        df[col] = df[col].dt.date
                elif dtype == "int":
                    df[col] = pd.to_numeric(df[col], errors="coerce").astype("Int64")
                elif dtype == "float":
                    df[col] = pd.to_numeric(df[col], errors="coerce").astype(float)
                elif dtype == "boolean":
                    df[col] = df[col].astype(str).str.lower().isin(['true', '1', 't', 'y', 'yes'])
                else:
                    df[col] = df[col].astype(dtype)
            except Exception as exc:
                logger.error("Cast failed for column '%s' to '%s': %s", col, dtype, exc)
        return df

    def _action_alias(self, df: pd.DataFrame, rule: TransformationRule) -> pd.DataFrame:
        """Create a copy of a column with a new name (alias)."""
        source_col = rule.column
        alias_col = rule.params.get("alias")
        if source_col and alias_col and source_col in df.columns:
            df[alias_col] = df[source_col].copy()
        return df

    # ── Column operations ─────────────────────────

    def _action_rename(self, df: pd.DataFrame, rule: TransformationRule) -> pd.DataFrame:
        old_name = rule.column
        new_name = rule.params.get("new_name")
        if old_name and new_name and old_name in df.columns:
            df = df.rename(columns={old_name: new_name})
        return df

    def _action_drop(self, df: pd.DataFrame, rule: TransformationRule) -> pd.DataFrame:
        col = rule.column
        columns = rule.params.get("columns", [])
        if col:
            columns.append(col)
        existing = [c for c in columns if c in df.columns]
        if existing:
            df = df.drop(columns=existing)
        return df

    def _action_add(self, df: pd.DataFrame, rule: TransformationRule) -> pd.DataFrame:
        """Add a derived column using a pandas eval expression."""
        new_column = rule.params.get("new_column")
        expression = rule.params.get("expression")
        if new_column and expression:
            try:
                df[new_column] = df.eval(expression)
            except Exception as exc:
                logger.error("Failed to add column '%s' with expression '%s': %s",
                             new_column, expression, exc)
        return df

    def _action_field_mapping(self, df: pd.DataFrame, rule: TransformationRule) -> pd.DataFrame:
        """Apply final field mappings: select, duplicate, and rename columns.
        
        SMART LOGIC: If the target column already exists in the DataFrame 
        (due to an alias or transformation earlier in the pipeline), 
        we use that column as-is instead of re-copying from the original source.
        This allows the JSON configuration to show user-friendly names 
        (Source_Original -> Target) while functionally using the isolated derivation.
        """
        mapping = rule.params.get("mapping", [])
        if not mapping:
            # Fallback to 'mappings' or 'field_mappings' key for compatibility
            mapping = rule.params.get("mappings") or rule.params.get("field_mappings", [])
            
        if not mapping:
            return df
            
        new_df_data = {}
        ordered_cols = []
        
        for m in mapping:
            src = m.get("source_column")
            tgt = m.get("target_column")
            
            # If target column already exists and is different from source, prioritize it
            if tgt in df.columns and tgt != src:
                new_df_data[tgt] = df[tgt].copy()
                ordered_cols.append(tgt)
            elif src in df.columns:
                new_df_data[tgt] = df[src].copy()
                ordered_cols.append(tgt)
                
        return pd.DataFrame(new_df_data)[ordered_cols]

    def _action_substring(self, df: pd.DataFrame, rule: TransformationRule) -> pd.DataFrame:
        """Extract a substring or split part from a column based on mode."""
        col = rule.column
        if not col or col not in df.columns:
            return df
        
        mode = rule.params.get("mode", "custom")
        mask = df[col].notna()
        
        if mode == "custom":
            start = rule.params.get("start", 0)
            length = rule.params.get("length", 10)
            df.loc[mask, col] = df.loc[mask, col].astype(str).str.slice(start=start, stop=start + length)
        elif mode == "first_n":
            n = rule.params.get("n", 5)
            df.loc[mask, col] = df.loc[mask, col].astype(str).str.slice(stop=n)
        elif mode == "last_n":
            n = rule.params.get("n", 5)
            # Use negative indexing to get last n characters
            df.loc[mask, col] = df.loc[mask, col].astype(str).str.slice(start=-n)
        elif mode == "before_delim":
            delim = rule.params.get("delimiter", "@")
            df.loc[mask, col] = df.loc[mask, col].astype(str).str.split(delim, n=1).str[0]
        elif mode == "after_delim":
            delim = rule.params.get("delimiter", "@")
            df.loc[mask, col] = df.loc[mask, col].astype(str).str.split(delim, n=1).str[1]
            
        return df

    def _action_row_number(self, df: pd.DataFrame, rule: TransformationRule) -> pd.DataFrame:
        """Sequential row number generator preserving continuity across chunks."""
        col = rule.column or "row_num"
        start_val = self._sequence_counters.get(col, 0)
        
        # Create sequence from start_val+1 to start_val+len(df)
        df[col] = range(start_val + 1, start_val + len(df) + 1)
        
        # Update counter
        self._sequence_counters[col] = start_val + len(df)
        return df

    def _action_surrogate_key(self, df: pd.DataFrame, rule: TransformationRule) -> pd.DataFrame:
        """Surrogate key logic: MAX + 1 across chunks."""
        col = rule.column or "surrogate_key"

        # LOAD_ID is a batch identifier: increment once per run and stamp all rows equally.
        if col.lower() == "load_id":
            if col not in self._batch_sequence_values:
                next_val = self._sequence_counters.get(col, 0) + 1
                self._batch_sequence_values[col] = next_val
                self._sequence_counters[col] = next_val
            df[col] = self._batch_sequence_values[col]
            return df

        # Other surrogate keys remain row-level sequences.
        return self._action_row_number(df, rule)

    def _action_load_metadata(self, df: pd.DataFrame, rule: TransformationRule) -> pd.DataFrame:
        """Inject current load_id and timestamp into the dataframe.
        
        If params include 'include_timestamp': True, adds LOAD_TIMESTAMP.
        """
        col = rule.column or "LOAD_ID"
        if self.load_id:
            df[col] = self.load_id
        
        if rule.params.get("include_timestamp", False) and self.load_timestamp:
            ts_col = rule.params.get("timestamp_column", "LOAD_TIMESTAMP")
            df[ts_col] = self.load_timestamp
            
        return df

    def _action_literal(self, df: pd.DataFrame, rule: TransformationRule) -> pd.DataFrame:
        """Assign a literal value to a column."""
        col = rule.column
        value = rule.params.get("value")
        data_type = rule.params.get("data_type")
        
        if col:
            # Handle data type conversion
            if data_type:
                if data_type == "integer":
                    if value == "" or value is None:
                        value = None
                    else:
                        try:
                            value = int(value)
                        except (ValueError, TypeError):
                            value = None
                elif data_type == "float":
                    if value == "" or value is None:
                        value = None
                    else:
                        try:
                            value = float(value)
                        except (ValueError, TypeError):
                            value = None
                elif data_type == "date":
                    if value == "" or value is None:
                        value = None
                    else:
                        try:
                            value = pd.to_datetime(value).date()
                        except (ValueError, TypeError):
                            value = None
                elif data_type == "datetime":
                    if value == "" or value is None:
                        value = None
                    else:
                        try:
                            value = pd.to_datetime(value)
                        except (ValueError, TypeError):
                            value = None
                # For string and other types, keep as is
            
            df[col] = value
        return df

    def _action_current_date(self, df: pd.DataFrame, rule: TransformationRule) -> pd.DataFrame:
        """Assign current date/time to a column."""
        col = rule.column
        if col:
            df[col] = pd.Timestamp.now()
        return df

    def _action_if_then_else(self, df: pd.DataFrame, rule: TransformationRule) -> pd.DataFrame:
        """Conditional if-then-else logic: result = trueValue if condition else falseValue."""
        col = rule.column
        condition = rule.params.get("condition")
        true_val = rule.params.get("trueValue")
        false_val = rule.params.get("falseValue")
        
        if col and condition:
            try:
                # Use pandas eval to evaluate the condition
                mask = df.eval(condition)
                df[col] = np.where(mask, true_val, false_val)
            except Exception as exc:
                logger.error("If-then-else failed for column '%s': %s", col, exc)
        return df

    def _action_data_masking(self, df: pd.DataFrame, rule: TransformationRule) -> pd.DataFrame:
        """Mask sensitive column data."""
        col = rule.column
        mask_type = rule.params.get("masking_type", "full")
        
        if not col or col not in df.columns:
            return df
            
        if mask_type == "full":
            df[col] = "****"
        elif mask_type == "hash":
            df[col] = df[col].astype(str).apply(lambda x: hashlib.md5(x.encode()).hexdigest() if pd.notnull(x) else x)
        elif mask_type == "null":
            df[col] = None
        elif mask_type == "partial":
            sf = rule.params.get("show_first", 2)
            sl = rule.params.get("show_last", 2)
            def partial_mask(val):
                if pd.isnull(val): return val
                s = str(val)
                if len(s) <= (sf + sl): return "****"
                return s[:sf] + "****" + s[-sl:]
            df[col] = df[col].apply(partial_mask)
        elif mask_type == "base64":
            df[col] = df[col].astype(str).apply(lambda x: base64.b64encode(x.encode()).decode() if pd.notnull(x) else x)
            
        return df

    def _action_data_unmasking(self, df: pd.DataFrame, rule: TransformationRule) -> pd.DataFrame:
        """Unmask (decode) column data."""
        col = rule.column
        unmask_type = rule.params.get("unmasking_type", "base64")
        
        if not col or col not in df.columns:
            return df
            
        if unmask_type == "base64":
            def safe_decode(x):
                try: return base64.b64decode(str(x)).decode() if pd.notnull(x) else x
                except: return x
            df[col] = df[col].apply(safe_decode)
            
        return df


    # ── Row transforms ────────────────────────────

    def _action_filter(self, df: pd.DataFrame, rule: TransformationRule) -> pd.DataFrame:
        """Filter rows using a pandas query expression.
        
        Tries to convert SQL-style 'IS NOT NULL' to pandas '.notnull()'
        """
        condition = rule.params.get("condition")
        if condition:
            # Basic SQL to Pandas translation for common null checks
            cond_prep = condition.replace(" IS NOT NULL", ".notnull()")
            cond_prep = cond_prep.replace(" IS NULL", ".isnull()")
            try:
                before = len(df)
                df = df.query(cond_prep, engine="python")
                logger.debug("Filter '%s' -> '%s': %d -> %d rows", condition, cond_prep, before, len(df))
            except Exception as exc:
                logger.error("Filter failed for condition '%s': %s", condition, exc)
        return df

    def _action_if(self, df: pd.DataFrame, rule: TransformationRule) -> pd.DataFrame:
        """Row-level conditional update."""
        col = rule.column
        condition = rule.params.get("condition")
        value = rule.params.get("value")
        
        if col and col in df.columns and condition:
            try:
                mask = df.eval(condition)
                df.loc[mask, col] = value
            except Exception as exc:
                logger.error("Row-level IF failed for column '%s': %s", col, exc)
        return df

    def _action_sort(self, df: pd.DataFrame, rule: TransformationRule) -> pd.DataFrame:
        """Sort rows by one or more columns."""
        cols = rule.params.get("columns", [])
        ascending = rule.params.get("ascending", True)
        if cols:
            existing = [c for c in cols if c in df.columns]
            if existing:
                df = df.sort_values(by=existing, ascending=ascending)
        return df

    def _action_sample(self, df: pd.DataFrame, rule: TransformationRule) -> pd.DataFrame:
        """Randomly sample rows."""
        fraction = rule.params.get("fraction", 0.1)
        return df.sample(frac=fraction)

    def _action_distinct(self, df: pd.DataFrame, rule: TransformationRule) -> pd.DataFrame:
        """Alias for deduplicate with no subset."""
        return self._action_deduplicate(df, rule)


    def _action_lookup(self, df: pd.DataFrame, rule: TransformationRule) -> pd.DataFrame:
        """Perform a lookup/join against another data source."""
        operation = rule.params.get("operation")
        
        if operation == "surrogate_key":
            # Handle surrogate key generation
            return self._action_surrogate_key(df, rule)
        elif operation == "join":
            lookup_table = rule.params.get("lookup_table")
            join_keys = rule.params.get("join_keys", [])
            join_type = rule.params.get("join_type", "left")
            
            # This requires a registry or some way to get the other dataframe
            # For now, we log a warning as cross-table joins in a single-stream Pandas transformer
            # need to be handled by the pipeline orchestrator or a specific LookupHandler.
            logger.warning("Lookup JOIN for '%s' requested - orchestrator should handle multi-source joins", lookup_table)
            return df
        else:
            logger.warning("Lookup operation '%s' not implemented yet - skipping", operation)
            return df
            
    def _action_scd_type_2(self, df: pd.DataFrame, rule: TransformationRule) -> pd.DataFrame:
        """Boilerplate SCD Type 2 columns."""
        eff_start = rule.params.get("effective_start", "eff_start_dt")
        eff_end = rule.params.get("effective_end", "eff_end_dt")
        is_current = rule.params.get("current_flag", "is_current")
        op_type = rule.params.get("operation_flag", "op_type")
        
        df[eff_start] = pd.Timestamp.now()
        df[eff_end] = pd.NaT
        df[is_current] = True
        df[op_type] = "INSERT"
        
        return df

    def _action_deduplicate(self, df: pd.DataFrame, rule: TransformationRule) -> pd.DataFrame:
        subset = rule.params.get("subset")  # list of columns or None for all
        keep = rule.params.get("keep", "first")
        before = len(df)
        df = df.drop_duplicates(subset=subset, keep=keep)
        logger.debug("Deduplicate (subset=%s): %d → %d rows", subset, before, len(df))
        return df

    # ── Aggregation ───────────────────────────────

    def _action_aggregate(self, df: pd.DataFrame, rule: TransformationRule) -> pd.DataFrame:
        """Group-by aggregation.

        params = {"group_by": ["dept"], "agg": {"salary": "sum", "id": "count"}}
        """
        group_by = rule.params.get("group_by", [])
        agg = rule.params.get("agg", {})
        if group_by and agg:
            try:
                df = df.groupby(group_by, as_index=False).agg(agg)
                logger.debug("Aggregated by %s: %d rows", group_by, len(df))
            except Exception as exc:
                logger.error("Aggregation failed: %s", exc)
        return df

    def _action_cdc_compare(self, df: pd.DataFrame, rule: TransformationRule) -> pd.DataFrame:
        """Compare incoming data with existing data and set CDC operation types."""
        op_column = rule.params.get("op_column")
        target_type = rule.params.get("target_type", "DB")
        target_path = rule.params.get("target_path", "")
        db_type = rule.params.get("db_type")
        schema = rule.params.get("schema", "public")
        connection_string = rule.params.get("connection_string")
        key_columns = rule.params.get("key_columns", [])
        compare_columns = rule.params.get("compare_columns", [])
        exclude_columns = rule.params.get("exclude_columns", [])
        
        if not op_column:
            logger.debug("CDC compare: no op_column specified, skipping")
            return df
            
        if not key_columns:
            logger.warning("CDC compare: no key_columns specified, cannot perform comparison")
            return df
            
        # For database targets, query existing data
        if target_type.upper() == "DB" and connection_string and db_type:
            try:
                from connectors.generic_connector import GenericConnector
                
                # Resolve environment variables / aliases gracefully from JSON Configs
                if connection_string and "://" not in connection_string:
                    import os
                    connection_string = os.getenv(connection_string, connection_string)
                
                # Create connector and get existing data
                connector = GenericConnector(db_type=db_type, connection_url=connection_string)
                engine = connector.get_engine()
                
                # Build query to get existing data with key columns and compare columns
                table_name = target_path
                if schema and schema != "public":
                    table_name = f"{schema}.{target_path}"
                
                select_cols = list(set(key_columns + compare_columns))
                
                keys_in_chunk = [str(x).replace("'", "''") for x in df[key_columns[0]].dropna().unique()]
                in_clause = ", ".join(f"'{x}'" for x in keys_in_chunk) if keys_in_chunk else "NULL"
                
                select_cols_str = ", ".join([f'"{c}"' for c in select_cols])
                query = f'SELECT {select_cols_str} FROM {table_name} WHERE "{key_columns[0]}" IN ({in_clause})'
                
                logger.debug("CDC compare: querying existing data from %s with chunk filter", table_name)
                existing_df = pd.read_sql(query, engine)
                
                if existing_df.empty:
                    # No existing data, all records are inserts
                    df[op_column] = "I"
                    logger.debug("CDC compare: no existing data, set %d rows as INSERT", len(df))
                    return df
                
                # Create composite key for comparison
                key_cols_str = key_columns
                df['_composite_key'] = df[key_cols_str].astype(str).agg('-'.join, axis=1)
                existing_df['_composite_key'] = existing_df[key_cols_str].astype(str).agg('-'.join, axis=1)
                # Deduplicate safely before merge to prevent pandas Cartesian explosion if target DB is dirty
                if not existing_df.empty:
                    existing_df = existing_df.drop_duplicates(subset=['_composite_key'], keep='last')
                
                # Set default operation as INSERT
                df[op_column] = "I"
                
                # Check for updates: records that exist in both but have different compare columns
                if compare_columns:
                    # Merge on composite key to find matches
                    merged = df.merge(existing_df, on='_composite_key', how='left', suffixes=('', '_existing'))
                    merged.index = df.index  # Align boolean mask indexes to prevent unhashable Series crash
                    
                    # For each compare column, check if values differ
                    record_exists = merged['_composite_key'].notna()  # Record exists
                    any_col_diff = pd.Series(False, index=merged.index)
                    
                    for col in compare_columns:
                        existing_col = f"{col}_existing"
                        if existing_col in merged.columns:
                            # Smart Numeric Tolerance to prevent precision drift (e.g. Pandas 609588.099 vs Postgres 609588.1)
                            if pd.api.types.is_numeric_dtype(merged[col]) or pd.api.types.is_numeric_dtype(merged[existing_col]):
                                col_num = pd.to_numeric(merged[col], errors='coerce').round(2)
                                exist_num = pd.to_numeric(merged[existing_col], errors='coerce').round(2)
                                col_diff = (col_num != exist_num) & ~(col_num.isna() & exist_num.isna())
                            else:
                                # String exact checks
                                col_s = merged[col].astype(str).str.replace(r'\.0$', '', regex=True)
                                col_s = col_s.replace('nan', '')
                                
                                exist_s = merged[existing_col].astype(str).str.replace(r'\.0$', '', regex=True)
                                exist_s = exist_s.replace('nan', '')
                                
                                col_diff = (col_s != exist_s)
                            
                            # Only one column needs to be different to consider it an update
                            any_col_diff = any_col_diff | col_diff
                    
                    update_mask = record_exists & any_col_diff
                    
                    # Set updates
                    df.loc[update_mask, op_column] = "U"
                    
                    inserts = (df[op_column] == "I").sum()
                    updates = (df[op_column] == "U").sum()
                    logger.debug("CDC compare: %d INSERT, %d UPDATE", inserts, updates)
                else:
                    logger.debug("CDC compare: no compare_columns specified, all existing records treated as inserts")
                
                # Clean up temporary column
                df.drop('_composite_key', axis=1, inplace=True)
                
            except Exception as exc:
                logger.error("CDC compare failed: %s", exc)
                # Fallback: treat all as inserts
                df[op_column] = "I"
        else:
            # For non-database targets or missing connection info, assume all inserts
            logger.debug("CDC compare: non-DB target or missing connection, assuming all inserts")
            df[op_column] = "I"
        
        return df
