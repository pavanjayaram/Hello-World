"""
DATABASE to DATABASE Pipeline.
"""

from __future__ import annotations

from config.models import LoadStrategy, TableConfig, TransformationRule
from connectors.generic_connector import GenericConnector
from connectors.oracle_connector import OracleConnector
from connectors.teradata_connector import TeradataConnector
from extractors.db_extractor import DBExtractor
from loaders.cdc_type2_loader import CDCType2Loader
from loaders.truncate_loader import TruncateLoader
from sqlalchemy import text
from transformers.pandas_transformer import PandasTransformer
from utils.logger import get_logger
from utils.monitor import JobMonitor

logger = get_logger("db_to_db")


def get_connector(db_cfg):
    db_type = db_cfg.db_type.value.lower()
    kwargs = db_cfg.model_dump()
    if db_type == "oracle":
        return OracleConnector(**kwargs)
    elif db_type == "teradata":
        return TeradataConnector(**kwargs)
    else:
        return GenericConnector(**kwargs)


def run(table_config: TableConfig, monitor: JobMonitor, run_id: str = None, run_timestamp: str = None) -> None:
    """Execute a DB → DB pipeline."""
    t_metrics = monitor.get_table(table_config.table_id)
    t_metrics.start()

    logger.info("Starting DB_TO_DB for table_id: %s (run_id: %s)", table_config.table_id, run_id)
    source_connector = None
    target_connector = None

    try:
        source_cfg = table_config.source.db
        target_cfg = table_config.target.db
        if not source_cfg or not target_cfg:
            raise ValueError("DB_TO_DB pipeline requires DB source and Target configs.")

        transformer = PandasTransformer()

        # Set background run metadata
        if run_id:
            transformer.set_load_metadata(run_id, run_timestamp)

        source_connector = get_connector(source_cfg)
        target_connector = get_connector(target_cfg)
        
        source_engine = source_connector.get_engine()
        target_engine = target_connector.get_engine()
        
        extractor = DBExtractor(engine=source_engine, table=source_cfg.table, query=source_cfg.query, schema=getattr(source_cfg, 'schema_name', None))


        # --- Handle Surrogate Key Logic (MAX + 1) ---
        for rule in table_config.transformations:
            if rule.action.lower() == "surrogate_key" or (rule.action.lower() == "lookup" and rule.params.get("operation") == "surrogate_key"):
                col = rule.column or rule.params.get("target_column")
                logger.info("Initializing surrogate_key for column '%s' — fetching current MAX from %s", col, target_cfg.table)
                try:
                    schema_prefix = f'"{target_cfg.schema_name}".' if target_cfg.schema_name else ""
                    query = f'SELECT MAX("{col}") FROM {schema_prefix}"{target_cfg.table}"'
                    with target_engine.connect() as conn:
                        res = conn.execute(text(query)).scalar()
                        curr_max = int(res) if res is not None else 0
                    transformer.set_sequence_start(col, curr_max)
                except Exception as e:
                    logger.info("Could not fetch MAX from DB for '%s', defaulting to 0: %s", col, e)
                    transformer.set_sequence_start(col, 0)

        if target_cfg.load_strategy == LoadStrategy.CDC_TYPE2:
            if not target_cfg.cdc_config:
                raise ValueError("CDC Type 2 strategy requires cdc_config")
            loader = CDCType2Loader(
                engine=target_engine,
                table=target_cfg.table,
                schema=target_cfg.schema_name,
                config=target_cfg.cdc_config,
                batch_size=target_cfg.batch_size
            )
        else:
            loader = TruncateLoader(
                engine=target_engine,
                table=target_cfg.table,
                schema=target_cfg.schema_name,
                batch_size=target_cfg.batch_size
            )

        chunk_size = table_config.chunk_size

        for chunk_idx, df_chunk in enumerate(extractor.extract(chunk_size=chunk_size)):
            t_metrics.chunks_processed += 1
            t_metrics.rows_extracted += len(df_chunk)

            if not df_chunk.empty and table_config.transformations:
                df_chunk = transformer.transform(df_chunk, table_config.transformations)
            
            t_metrics.rows_transformed += len(df_chunk)

            if not df_chunk.empty:
                load_metrics = loader.load(df_chunk)
                t_metrics.rows_loaded += load_metrics.get("rows_loaded", 0)
                t_metrics.rows_inserted += load_metrics.get("rows_inserted", 0)
                t_metrics.rows_updated += load_metrics.get("rows_updated", 0)
                t_metrics.rows_unchanged += load_metrics.get("rows_unchanged", 0)

        # 4. Finalize post-load logic (e.g. Full Volume Soft Deletes)
        if hasattr(loader, "finalize"):
            try:
                final_metrics = loader.finalize()
                deleted = final_metrics.get("rows_deleted", 0)
                if deleted > 0:
                    t_metrics.rows_updated += deleted
                    t_metrics.rows_loaded += deleted
            except Exception as e:
                logger.error("Error during loader finalization: %s", e)

        t_metrics.finish()
        logger.info("Completed DB_TO_DB for table_id: %s", table_config.table_id)

    except Exception as exc:
        t_metrics.add_error(str(exc))
        t_metrics.finish("failed")
        raise
    finally:
        if source_connector:
            source_connector.dispose()
        if target_connector:
            target_connector.dispose()
