"""
Core dispatch engine for executing ETL pipelines based on JSON config.
"""

from __future__ import annotations

import json
from datetime import datetime
from pathlib import Path

from config.mappers import map_frontend_to_job_config
from config.models import JobConfig
from core.parallel import run_tables_concurrent
import pipelines.db_to_db as db_to_db
import pipelines.db_to_file as db_to_file
import pipelines.file_to_db as file_to_db
import pipelines.file_to_file as file_to_file
from utils.logger import get_logger, setup_logger
from utils.monitor import JobMonitor

# Registry maps PipelineType enum values to the runner function
PIPELINE_REGISTRY = {
    "FILE_TO_FILE": file_to_file.run,
    "FILE_TO_DB": file_to_db.run,
    "DB_TO_DB": db_to_db.run,
    "DB_TO_FILE": db_to_file.run,
}

def load_config(config_path: str | Path) -> JobConfig:
    """Parse JobConfig from a JSON file."""
    with open(config_path, "r", encoding="utf-8") as f:
        data = json.load(f)
    if "tables" in data and "job_id" in data:
        return JobConfig(**data)
    return map_frontend_to_job_config(data)

def execute_job(config: JobConfig) -> JobMonitor:
    """Run an entire job configuration."""
    setup_logger(job_id=config.job_id)
    logger = get_logger("engine")
    logger.info("Executing job '%s' (Pipeline: %s) with %d tables",
                config.job_id, config.pipeline_type.value, len(config.tables))

    runner = PIPELINE_REGISTRY.get(config.pipeline_type.value)
    if not runner:
        raise ValueError(f"Unknown pipeline type: {config.pipeline_type}")

    monitor = JobMonitor(job_id=config.job_id)
    run_timestamp = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    run_id = f"{config.job_id}_{datetime.now().strftime('%Y%m%d_%H%M%S')}"

    try:
        if config.max_workers > 1 and len(config.tables) > 1:
            logger.info("Running %d workers for concurrency", config.max_workers)
            run_tables_concurrent(config.tables, runner, monitor, max_workers=config.max_workers)
        else:
            logger.info("Running sequentially")
            for table_cfg in config.tables:
                try:
                    runner(table_cfg, monitor, run_id=run_id, run_timestamp=run_timestamp)
                except Exception as e:
                    logger.error("Table %s failed: %s", table_cfg.table_id, e)
                    # We continue to the next table even if one fails
    finally:
        monitor.finish(status="failed" if any(t.status == "failed" for t in monitor.tables.values()) else "success")
        monitor.print_summary()
        monitor.write_audit()

    return monitor
