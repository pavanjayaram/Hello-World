"""
Pydantic configuration models for the ETL engine.

Defines the schema for job configurations, including source/target definitions,
transformation rules, and load strategies.
"""

from __future__ import annotations

from enum import Enum
from typing import Any, Dict, List, Optional

from pydantic import BaseModel, Field, model_validator


# ──────────────────────────────────────────────
# Enums
# ──────────────────────────────────────────────

class SourceType(str, Enum):
    FILE = "FILE"
    DB = "DB"


class TargetType(str, Enum):
    FILE = "FILE"
    DB = "DB"


class FileFormat(str, Enum):
    CSV = "csv"
    TSV = "tsv"
    TXT = "txt"
    EXCEL = "excel"
    PARQUET = "parquet"
    JSON = "json"
    AVRO = "avro"


class DBType(str, Enum):
    ORACLE = "oracle"
    TERADATA = "teradata"
    POSTGRES = "postgres"
    MYSQL = "mysql"
    SQLSERVER = "sqlserver"
    SQLITE = "sqlite"


class LoadStrategy(str, Enum):
    TRUNCATE_LOAD = "truncate_load"
    CDC_TYPE2 = "cdc_type2"
    APPEND = "append"


class PipelineType(str, Enum):
    FILE_TO_FILE = "FILE_TO_FILE"
    FILE_TO_DB = "FILE_TO_DB"
    DB_TO_DB = "DB_TO_DB"
    DB_TO_FILE = "DB_TO_FILE"


# ──────────────────────────────────────────────
# Source / Target configs
# ──────────────────────────────────────────────

class FileSourceConfig(BaseModel):
    """Config for file-based sources."""
    format: FileFormat
    path: str
    delimiter: str = ","
    header: bool = True
    encoding: str = "utf-8"
    sheet_name: Optional[str] = None  # For Excel files


class DBSourceConfig(BaseModel):
    """Config for database sources."""
    db_type: DBType
    connection_url: Optional[str] = Field(None, alias="connectionString")  # Resolve alias from frontend
    host: Optional[str] = None
    port: Optional[int] = None
    database: Optional[str] = None
    user: Optional[str] = None
    password: Optional[str] = None
    service_name: Optional[str] = None  # Oracle
    schema_name: Optional[str] = Field(None, alias="schema")
    table: Optional[str] = None
    query: Optional[str] = None  # Custom SQL
    
    class Config:
        populate_by_name = True


class SourceConfig(BaseModel):
    """Unified source configuration."""
    type: SourceType
    file: Optional[FileSourceConfig] = None
    db: Optional[DBSourceConfig] = None


class FileTargetConfig(BaseModel):
    """Config for file-based targets."""
    format: FileFormat
    path: str
    delimiter: str = ","
    header: bool = True
    encoding: str = "utf-8"
    mode: str = "overwrite"  # overwrite | append


class CDCType2Config(BaseModel):
    """SCD Type 2 specific settings."""
    natural_keys: List[str]  # Business key columns
    tracked_columns: List[str]  # Columns to track for changes
    effective_date_col: str = "effective_date"
    end_date_col: str = "end_date"
    end_date_default: Optional[Any] = None
    is_current_col: str = "is_current"
    current_value: Optional[str] = None
    inactive_value: Optional[str] = None
    hash_col: Optional[str] = "row_hash"
    op_column: Optional[str] = None


class DBTargetConfig(BaseModel):
    """Config for database targets."""
    db_type: DBType
    connection_url: Optional[str] = Field(None, alias="connectionString")  # Resolve alias
    host: Optional[str] = None
    port: Optional[int] = None
    database: Optional[str] = None
    user: Optional[str] = None
    password: Optional[str] = None
    service_name: Optional[str] = None
    schema_name: Optional[str] = None
    table: str
    load_strategy: LoadStrategy = LoadStrategy.TRUNCATE_LOAD
    cdc_config: Optional[CDCType2Config] = None
    batch_size: int = 1000  # Default to 1000 for standard loads
    
    class Config:
        populate_by_name = True


class TargetConfig(BaseModel):
    """Unified target configuration."""
    type: TargetType
    file: Optional[FileTargetConfig] = None
    db: Optional[DBTargetConfig] = None


# ──────────────────────────────────────────────
# Transformations
# ──────────────────────────────────────────────

class TransformationRule(BaseModel):
    """Single transformation rule."""
    action: str  # upper, lower, trim, cast, rename, add, drop, filter, deduplicate, aggregate
    column: Optional[str] = None
    params: Dict[str, Any] = Field(default_factory=dict)
    # Examples:
    #   {"action": "upper", "column": "name"}
    #   {"action": "cast", "column": "salary", "params": {"dtype": "float"}}
    #   {"action": "rename", "column": "old_name", "params": {"new_name": "new_col"}}
    #   {"action": "add", "params": {"new_column": "full_name", "expression": "first + ' ' + last"}}
    #   {"action": "filter", "params": {"condition": "salary > 50000"}}
    #   {"action": "deduplicate", "params": {"subset": ["id", "name"]}}
    #   {"action": "aggregate", "params": {"group_by": ["dept"], "agg": {"salary": "sum"}}}

    @model_validator(mode="before")
    @classmethod
    def _merge_legacy_fields_into_params(cls, data: Any) -> Any:
        """Support both nested params and legacy top-level transformation fields."""
        if not isinstance(data, dict):
            return data

        params = dict(data.get("params") or {})
        reserved = {"action", "column", "params"}
        legacy_params = {k: v for k, v in data.items() if k not in reserved}

        if legacy_params:
            params = {**legacy_params, **params}
            data = {**data, "params": params}

        return data


# ──────────────────────────────────────────────
# Table / Job configs
# ──────────────────────────────────────────────

class TableConfig(BaseModel):
    """Configuration for a single table/dataset in the pipeline."""
    table_id: str
    source: SourceConfig
    target: TargetConfig
    transformations: List[TransformationRule] = Field(default_factory=list)
    chunk_size: int = 50000
    use_dask: bool = False


class JobConfig(BaseModel):
    """Top-level job configuration."""
    job_id: str
    pipeline_type: PipelineType
    tables: List[TableConfig]
    max_workers: int = 1  # Number of concurrent table processes
    description: str = ""
    mapping_file: Optional[str] = None  # Path to Excel mapping file
