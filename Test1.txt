"""
Helpers for translating legacy frontend pipeline JSON into engine JobConfig.
"""

from __future__ import annotations

import uuid

from config.models import JobConfig


def map_frontend_to_job_config(data: dict) -> JobConfig:
    """Translate the legacy UI flat pipeline JSON into the engine JobConfig."""
    if "tables" in data and "job_id" in data:
        return JobConfig(**data)

    p_info = data.get("pipeline", {})
    job_id = p_info.get("pipeline_id", str(uuid.uuid4()))
    p_type = p_info.get("pipeline_type", "FILE_TO_FILE")

    src_data = data.get("source", {})
    tgt_data = data.get("target", {})
    mapping = data.get("field_mappings", [])
    transforms = data.get("transformations", [])

    src_type = src_data.get("type", "FILE").upper()
    if src_type == "DB":
        source = {
            "type": "DB",
            "db": {
                "db_type": src_data.get("db_type", "sqlite").lower(),
                "connection_url": src_data.get("connectionString"),
                "table": src_data.get("table"),
                "schema": src_data.get("schema")
            }
        }
    else:
        source = {
            "type": "FILE",
            "file": {
                "format": src_data.get("format", "csv").lower(),
                "path": src_data.get("path", ""),
                "delimiter": src_data.get("options", {}).get("delimiter", ",")
            }
        }

    tgt_type = tgt_data.get("type", "FILE").upper()
    if tgt_type == "DB":
        is_cdc = any(t.get("type") == "cdc_compare" for t in (transforms or []))
        load_strat = "cdc_type2" if is_cdc else "truncate_load"

        target = {
            "type": "DB",
            "db": {
                "db_type": tgt_data.get("db_type", "sqlite").lower(),
                "connection_url": tgt_data.get("connectionString"),
                "table": tgt_data.get("table"),
                "load_strategy": load_strat,
                "schema_name": tgt_data.get("schema")
            }
        }
        if is_cdc:
            cdc_t = next((t for t in transforms if t.get("type") == "cdc_compare"), {})

            target_cols = [m.get("target_column", "").lower() for m in mapping]
            eff_col = next((c for c in target_cols if c in ("eff_date", "effective_date", "start_date")), "effective_date")
            end_col = next((c for c in target_cols if c in ("end_date", "expiry_date")), "end_date")
            curr_col = next((c for c in target_cols if c in ("active_flag", "is_current", "current_flag")), "is_current")

            case_map = {m.get("target_column", "").lower(): m.get("target_column") for m in mapping}
            eff_col = case_map.get(eff_col, eff_col)
            end_col = case_map.get(end_col, end_col)
            curr_col = case_map.get(curr_col, curr_col)

            hash_col = next((c for c in target_cols if c in ("row_hash", "hash_val", "md5_hash")), None)
            hash_col = case_map.get(hash_col, hash_col)

            literal_defaults = {
                t.get("column"): t.get("value")
                for t in transforms
                if t.get("action", t.get("type", "")).lower() == "literal" and t.get("column")
            }

            target["db"]["cdc_config"] = {
                "natural_keys": cdc_t.get("key_columns", []),
                "tracked_columns": cdc_t.get("compare_columns", []),
                "effective_date_col": eff_col,
                "end_date_col": end_col,
                "end_date_default": literal_defaults.get(end_col),
                "is_current_col": curr_col,
                "current_value": literal_defaults.get(curr_col),
                "hash_col": hash_col,
                "op_column": cdc_t.get("op_column")
            }
    else:
        opts = tgt_data.get("options", {})
        if isinstance(opts, list):
            opts = {}
        target = {
            "type": "FILE",
            "file": {
                "format": tgt_data.get("format", "parquet").lower(),
                "path": tgt_data.get("path", ""),
                "delimiter": opts.get("delimiter", ",") if isinstance(opts, dict) else ",",
                "mode": tgt_data.get("mode", "overwrite")
            }
        }

    out_transforms = []
    for t in transforms:
        action = t.get("action", t.get("type", "")).lower()
        if not action:
            continue
            
        p = {k: v for k, v in t.items() if k not in ("action", "type", "column")}
        out_transforms.append({
            "action": action,
            "column": t.get("column") or t.get("target_column"),
            "params": p
        })

    if mapping:
        out_transforms.append({
            "action": "field_mapping",
            "column": "",
            "params": {"mapping": mapping}
        })

    table_cfg = {
        "table_id": job_id + "_tbl",
        "source": source,
        "target": target,
        "transformations": out_transforms,
        "chunk_size": 10000
    }

    return JobConfig(job_id=job_id, pipeline_type=p_type, tables=[table_cfg])
