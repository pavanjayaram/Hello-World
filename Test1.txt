"""
Job monitoring and metrics collection.

Tracks extraction, transformation, and load metrics per table
and writes a summary JSON to the audit/ directory.
"""

from __future__ import annotations

import json
import os
import time
from dataclasses import dataclass, field, asdict
from datetime import datetime
from pathlib import Path
from typing import Any, Dict, List, Optional

from utils.logger import get_logger

logger = get_logger("monitor")


@dataclass
class TableMetrics:
    """Metrics for a single table in the pipeline."""
    table_id: str
    status: str = "pending"  # pending | running | success | failed
    rows_extracted: int = 0
    rows_transformed: int = 0
    rows_loaded: int = 0
    rows_inserted: int = 0
    rows_updated: int = 0
    rows_unchanged: int = 0
    rows_rejected: int = 0
    chunks_processed: int = 0
    errors: List[str] = field(default_factory=list)
    start_time: Optional[float] = None
    end_time: Optional[float] = None

    @property
    def elapsed_seconds(self) -> float:
        if self.start_time is None:
            return 0.0
        end = self.end_time or time.time()
        return round(end - self.start_time, 2)

    def start(self) -> None:
        self.status = "running"
        self.start_time = time.time()

    def finish(self, status: str = "success") -> None:
        self.status = status
        self.end_time = time.time()

    def add_error(self, error: str) -> None:
        self.errors.append(error)
        logger.error("[%s] %s", self.table_id, error)

    def to_dict(self) -> Dict[str, Any]:
        d = asdict(self)
        d["elapsed_seconds"] = self.elapsed_seconds
        return d


class JobMonitor:
    """Aggregates metrics for an entire ETL job."""

    def __init__(self, job_id: str, audit_dir: str | None = None):
        self.job_id = job_id
        self.start_time = time.time()
        self.end_time: Optional[float] = None
        self.status: str = "running"
        self.tables: Dict[str, TableMetrics] = {}

        if audit_dir is None:
            audit_dir = os.path.join(
                os.path.dirname(os.path.dirname(__file__)), "audit"
            )
        self.audit_dir = audit_dir
        Path(self.audit_dir).mkdir(parents=True, exist_ok=True)

    def get_table(self, table_id: str) -> TableMetrics:
        """Get or create metrics for a table."""
        if table_id not in self.tables:
            self.tables[table_id] = TableMetrics(table_id=table_id)
        return self.tables[table_id]

    @property
    def elapsed_seconds(self) -> float:
        end = self.end_time or time.time()
        return round(end - self.start_time, 2)

    def finish(self, status: str = "success") -> None:
        self.status = status
        self.end_time = time.time()

    def summary(self) -> Dict[str, Any]:
        """Generate a full summary dict."""
        return {
            "job_id": self.job_id,
            "status": self.status,
            "started_at": datetime.fromtimestamp(self.start_time).isoformat(),
            "finished_at": (
                datetime.fromtimestamp(self.end_time).isoformat()
                if self.end_time else None
            ),
            "elapsed_seconds": self.elapsed_seconds,
            "tables": {
                tid: tm.to_dict() for tid, tm in self.tables.items()
            },
        }

    def write_audit(self) -> str:
        """Write metrics JSON to audit directory. Returns the file path."""
        timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
        filename = f"{self.job_id}_{timestamp}.json"
        filepath = os.path.join(self.audit_dir, filename)

        with open(filepath, "w", encoding="utf-8") as f:
            json.dump(self.summary(), f, indent=2, default=str)

        logger.info("Audit written: %s", filepath)
        return filepath

    def print_summary(self) -> None:
        """Log a human-readable summary."""
        s = self.summary()
        logger.info("=" * 60)
        logger.info("JOB SUMMARY: %s", s["job_id"])
        logger.info("Status: %s | Elapsed: %.2fs", s["status"], s["elapsed_seconds"])
        logger.info("-" * 60)
        for tid, tm in s["tables"].items():
            logger.info(
                "  %-20s | status=%-8s | extracted=%d | loaded=%d | elapsed=%.2fs",
                tid, tm["status"], tm["rows_extracted"], tm["rows_loaded"],
                tm["elapsed_seconds"],
            )
            if tm.get("rows_inserted") or tm.get("rows_updated"):
                logger.info(
                    "  %-20s | inserted=%d | updated=%d | unchanged=%d",
                    "", tm["rows_inserted"], tm["rows_updated"], tm["rows_unchanged"],
                )
            if tm["errors"]:
                for err in tm["errors"]:
                    logger.error("  %-20s | ERROR: %s", tid, err)
        logger.info("=" * 60)
