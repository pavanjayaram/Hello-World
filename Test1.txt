"""
Utility for resolving environment variables and connection string aliases.
"""

import os
from pathlib import Path
from dotenv import load_dotenv
from utils.logger import get_logger

logger = get_logger("env_utils")

# Ensure .env is loaded
env_path = Path(__file__).parent.parent / ".env"
if env_path.exists():
    logger.info("Found .env at %s, loading...", env_path)
    load_dotenv(env_path)
else:
    logger.warning(".env NOT found at %s", env_path)

def resolve_connection_string(conn_str: str) -> str:
    """
    Resolve a connection string alias (e.g., 'POSTGRES_URL') to its actual value.
    """
    if not conn_str:
        return ""
    
    # If it looks like a full URL already, don't resolve
    if "://" in conn_str or "host=" in conn_str or conn_str.startswith("sqlite"):
        return conn_str
    
    # Try fetching from environment (loaded via load_dotenv above)
    val = os.getenv(conn_str)
    if val:
        logger.debug("Resolved alias '%s' using environment variable.", conn_str)
        return val.strip('"\'')
    
    logger.warning("Could not resolve connection alias '%s' (not found in env).", conn_str)
    return conn_str
