import json
from pydantic import BaseModel, Field
from typing import Optional

class DBSourceConfig(BaseModel):
    schema_name: Optional[str] = Field(None, alias="schema")
    table: Optional[str] = None

data = {"schema": "company", "table": "employee"}
obj = DBSourceConfig(**data)
print("schema_name is:", obj.schema_name)
print("getattr schema_name is:", getattr(obj, "schema_name", None))
print("getattr schema is:", getattr(obj, "schema", None))
