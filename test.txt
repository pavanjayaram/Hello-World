def call_api(url, payload, soap_action):
    headers = {
        "Content-Type": "text/xml",
        "SOAPAction": soap_action
    }

    response = requests.post(url, data=payload, headers=headers, timeout=30, verify=get_ssl_config())

    fault = extract_soap_fault(response.text)

    if fault:
        print("\n🚨 SOAP FAULT:")
        print(fault)
        return None

    return response.text
