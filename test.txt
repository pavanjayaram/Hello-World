import sys
import json
import logging
import os
from pyspark.sql import SparkSession
from dotenv import load_dotenv

def main():
    if len(sys.argv) < 2:
        print(json.dumps({"error": "Missing config file path"}))
        sys.exit(1)
        
    config_path = sys.argv[1]
    
    # Load env matching the API
    backend_dir = os.path.dirname(os.path.abspath(__file__))
    project_root = os.path.dirname(backend_dir)
    load_dotenv(os.path.join(project_root, '.env'))
    try:
        with open(config_path, 'r') as f:
            config = json.load(f)
            
        db_type = config.get("db_type", "").lower()
        table = config.get("table")
        schema = config.get("schema", "public")
        connection_string = config.get("connectionString")
        
        if not table or not connection_string:
            print(json.dumps({"error": "Missing table or connection string"}))
            sys.exit(1)
            
        # Initialize Spark dynamically with the required drivers
        backend_dir = os.path.dirname(os.path.abspath(__file__))
        jars = [
            os.path.join(backend_dir, "jars", "ojdbc8.jar"),
            os.path.join(backend_dir, "jars", "postgresql-42.7.3.jar")
        ]
        
        # Suppress logging for faster startup
        logging.getLogger("py4j").setLevel(logging.ERROR)
        
        spark = SparkSession.builder \
            .appName("MetadataFetcher") \
            .config("spark.jars", ",".join(jars)) \
            .config("spark.ui.enabled", "false") \
            .getOrCreate()
            
        spark.sparkContext.setLogLevel("ERROR")
        
        # Build jdbc read properties
        properties = {}
        
        # Extract user/password if available in config.connection
        conn_details = config.get("connection", {})
        if conn_details.get("username"):
            properties["user"] = conn_details.get("username")
        if conn_details.get("password"):
            properties["password"] = conn_details.get("password")
            
        # Resolve environment variable if connection string is just a key like POSTGRES_URL
        if connection_string and connection_string in os.environ:
            connection_string = os.environ[connection_string]
            
        # Parse non-JDBC URIs using the shared DBSourceHandler
        if connection_string and not connection_string.startswith("jdbc:"):
            from sources import DBSourceHandler
            handler = DBSourceHandler()
            connection_string, parsed_props = handler._parse_connection_string(connection_string, db_type)
            for k, v in parsed_props.items():
                if k not in properties or not properties[k]:
                    properties[k] = v
        
        target_table = f"{schema}.{table}" if db_type == "postgres" and schema else table
        if db_type == "oracle" and schema:
            target_table = f"{schema}.{table}"
            
        # We only need the schema layout, we don't need any actual data
        # To do this incredibly fast, we formulate a pushdown query with 0 results.
        pushdown_query = f"(SELECT * FROM {target_table} WHERE 1=0) AS tmp"
        
        df = spark.read.jdbc(
            url=connection_string,
            table=pushdown_query,
            properties=properties
        )
        
        columns = df.columns
        print(json.dumps({"columns": columns}))
        
        spark.stop()
        
    except Exception as e:
        print(json.dumps({"error": str(e)}))
        sys.exit(1)

if __name__ == "__main__":
    main()
