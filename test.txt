def check_service(cfg):
    name = cfg.get("name", "UNKNOWN")
    ignore_if_down = cfg.get("ignore_if_down", True)

    env_config = cfg.get("environments", {}).get(ENV)

    if not env_config:
        return False, f"{name} -> No config for ENV {ENV}"

    lb_url = env_config.get("lb_url")
    nodes = env_config.get("nodes", [])

    failures = []
    success_msgs = []

    # =========================
    # 1. CHECK LOAD BALANCER
    # =========================
    if lb_url:
        status, msg = check_api({
            "name": f"{name}_LB",
            "url": lb_url,
            "ignore_if_down": False
        })

        log(msg)

        if not status:
            failures.append(msg)
        else:
            success_msgs.append(msg)

    # =========================
    # 2. CHECK NODES
    # =========================
    for i, node_url in enumerate(nodes, start=1):
        status, msg = check_api({
            "name": f"{name}_NODE{i}",
            "url": node_url,
            "ignore_if_down": ignore_if_down
        })

        log(msg)

        if not status:
            failures.append(msg)
        else:
            success_msgs.append(msg)

    # =========================
    # FINAL RESULT
    # =========================
    if failures:
        return False, "\n".join(failures)

    return True, f"{name} -> ALL HEALTHY"
