from pyspark.sql import DataFrame
from abc import ABC, abstractmethod
from typing import Dict
import logging
import os

class TargetHandler(ABC):
    @abstractmethod
    def write(self, df: DataFrame, config: Dict):
        pass

import glob
import shutil

class FileTargetHandler(TargetHandler):
    def write(self, df: DataFrame, config: Dict):
        format_type = config.get('format', 'csv').lower()
        path = config.get('path')
        mode = config.get('mode', 'overwrite')
        options = config.get('options', {})
        partitions = config.get("coalesce", 1) # Default to 1 partition for single file output
        
        if not path:
            raise ValueError("Target path is required for FILE type")
            
        # Spark writes directories, not files. We write to a temporary _spark_staging charade.
        temp_dir = path + "_spark_staging"
        
        df = df.coalesce(partitions)
        logging.info(f"Coalesced output to {partitions} partition(s).")
            
        logging.info(f"Writing FILE target to temporary dir {temp_dir} as {format_type} (mode: {mode})")
        writer = df.write.format(format_type).mode(mode)
        
        for k, v in options.items():
            writer = writer.option(k, str(v).lower() if isinstance(v, bool) else v)
            
        writer.save(temp_dir)
        
        # Post-Processing: Rename the messy part-00000 file to the exact Target Path requested
        try:
            # Find the actual data file (ignore _SUCCESS, .crc, etc.)
            part_files = glob.glob(os.path.join(temp_dir, "part-*"))
            
            if not part_files:
                logging.warning(f"No part-* files found in {temp_dir}. File may be completely empty.")
                return
                
            data_file = part_files[0] # Grab the first (and only) partition file
            
            # If the specific target file already exists and mode is overwrite, remove it
            if mode == 'overwrite' and os.path.exists(path):
                if os.path.isfile(path):
                    os.remove(path)
                elif os.path.isdir(path):
                    shutil.rmtree(path)
                    
            # Ensure the parent directory of the final path exists
            os.makedirs(os.path.dirname(path), exist_ok=True)
            
            # Move and rename the part file to the exact intended path
            shutil.move(data_file, path)
            logging.info(f"Successfully renamed Spark part file to exact target: {path}")
            
            # Cleanup the temporary Spark directory
            shutil.rmtree(temp_dir)
            
        except Exception as e:
            logging.error(f"Failed to cleanly rename Spark output file: {str(e)}")
            raise e

class DBTargetHandler(TargetHandler):
    def write(self, df: DataFrame, config: Dict):
        target_config = config.get('target', config)
        
        db_type = target_config.get('db_type', 'postgres')
        table = target_config.get('table')
        schema = target_config.get('schema', 'public')
        mode = target_config.get('mode', 'append')
        connection_string = target_config.get('connectionString')
        properties = {
            "user": target_config.get("user", ""),
            "password": target_config.get("password", ""),
            **target_config.get('options', {})
        }
        properties['driver'] = self._get_driver(db_type)
        
        if connection_string and connection_string in os.environ:
             connection_string = os.environ[connection_string]
             
        # Optional: Override properties if full JDBC URI parsing is requested (like in sources.py)
        if connection_string and not connection_string.startswith("jdbc:"):
            from sources import DBSourceHandler
            # Utilizing the parser from DBSourceHandler to extract user/pass from URI
            handler = DBSourceHandler()
            connection_string, parsed_props = handler._parse_connection_string(connection_string, db_type)
            properties.update(parsed_props)
             
        qualified_table = f"{schema}.{table}" if schema and db_type in ['postgres', 'sqlserver', 'oracle'] else table
        
        # Ensure driver is set in properties before any JDBC write operation
        properties['driver'] = self._get_driver(db_type)
        # Check if CDC Flag is present AND if a CDC transformation was actually run!
        cdc_col = next((c for c in df.columns if c in ['op_type', 'CDC_Flag']), None)
        
        # Verify the pipeline actually contained a cdc_compare transformation explicitly.
        # Otherwise, user might just have an arbitrary text column named "CDC_Flag" and expects a pure insert.
        is_cdc_pipeline = False
        exclude_cols = []
        transforms = config.get('transformations', []) if config else []
        for t in transforms:
            if t.get('type') in ['cdc', 'cdc_compare']:
                is_cdc_pipeline = True
                exclude_cols = t.get('exclude_columns', [])
                break
                
        # Ensure driver is set in properties before any JDBC write operation
        properties['driver'] = self._get_driver(db_type)
        
        if cdc_col and is_cdc_pipeline:
            logging.warning(f"SCD Type 2 CDC operations ({cdc_col}) explicitly detected. Executing Upsert Automation.")
            self._write_scd_type2_cdc(df, qualified_table, connection_string, properties, db_type, cdc_col, exclude_cols)
        else:
            logging.info(f"Writing DB target DIRECTLY to {qualified_table} as {mode} (CDC staging removed)")
            df.write.jdbc(url=connection_string, table=qualified_table, mode=mode, properties=properties)

    def _get_driver(self, db_type: str) -> str:
        driver_map = {
            'postgres': 'org.postgresql.Driver',
            'mysql': 'com.mysql.cj.jdbc.Driver',
            'oracle': 'oracle.jdbc.driver.OracleDriver',
            'sqlserver': 'com.microsoft.sqlserver.jdbc.SQLServerDriver'
        }
        return driver_map.get(db_type, 'org.postgresql.Driver')

    def _write_scd_type2_cdc(self, df: DataFrame, table: str, jdbc_url: str, properties: dict, db_type: str, op_col: str, exclude_cols: list = None):
        staging_table = f"{table}_staging"
        logging.info(f"Writing CDC payload to STAGING table: {staging_table}")
        
        # Write purely with Drop/Create Overwrite so the staging table is fresh for this batch
        properties['driver'] = self._get_driver(db_type)
        df.write.jdbc(url=jdbc_url, table=staging_table, mode="overwrite", properties=properties)
        
        logging.info(f"Executing Native JDBC MERGE from {staging_table} to {table} for SCD Type 2")
        
        # Extract Primary Key from properties or generic convention
        pk = properties.get('primary_key', 'EMP_ID') # Fallback to EMP_ID if not explicitly provided
        
        # Execute Native SQL via Py4J DriverManager using Properties to inherit Spark's Classloader
        spark = df.sparkSession
        jvm = spark._jvm
        
        # Pre-fetch table columns for dynamic resolution in SQL
        target_cols = spark.read.jdbc(url=jdbc_url, table=table, properties=properties).columns
        
        props = jvm.java.util.Properties()
        props.put("user", properties.get('user', ''))
        props.put("password", properties.get('password', ''))
        props.put("driver", properties['driver'])
        
        # Get connection natively
        conn = jvm.java.sql.DriverManager.getConnection(jdbc_url, props)
        stmt = conn.createStatement()
        
        try:
            # Step 1: Execute Deactivations (D rows). Update Target where Target PK = Staging PK and Staging Op = 'D'
            # Note: Hardcoded to Active_Flag and end_date per user requirement constraints.
            # We also update the operation flag to 'D' if it exists in the target.
            set_clause = "\"Active_Flag\" = 'N', \"end_date\" = CURRENT_DATE"
            if op_col in target_cols:
                set_clause += f", \"{op_col}\" = 'D'"
                
            update_sql = f"""
                UPDATE {table} t
                SET {set_clause}
                WHERE EXISTS (
                    SELECT 1 FROM {staging_table} s 
                    WHERE s."{pk}" = t."{pk}" AND s."{op_col}" = 'D'
                ) AND t."Active_Flag" = 'Y'
            """
            logging.info(f"Executing Deactivation: {update_sql.strip()}")
            deactivated_count = stmt.executeUpdate(update_sql)
            logging.info(f"SCD Type 2: Successfully Deactivated {deactivated_count} old records.")
            
            # Step 2: Execute Inserts (I rows).
            # We build the INSERT list by intersecting columns that exist in BOTH the staging table 
            # and the final target table.
            staging_cols = spark.read.jdbc(url=jdbc_url, table=staging_table, properties=properties).columns
            
            # Find intersection, maintaining order from staging if possible
            valid_cols = [c for c in staging_cols if c in target_cols]
            col_str = ", ".join([f'"{c}"' for c in valid_cols])
            
            insert_sql = f"""
                INSERT INTO {table} ({col_str})
                SELECT {col_str} FROM {staging_table}
                WHERE "{op_col}" IN ('I', 'U')
            """
            logging.info(f"Executing Insertion: {insert_sql.strip()}")
            inserted_count = stmt.executeUpdate(insert_sql)
            logging.info(f"SCD Type 2: Successfully Inserted {inserted_count} new records.")
            
            # Step 3: Cleanup Staging Table
            stmt.execute(f"DROP TABLE IF EXISTS {staging_table}")
            logging.info("SCD Type 2 Workflow Complete. Staging table dropped.")
            
        except Exception as e:
            logging.error(f"Failed to execute SCD Type 2 Merge: {str(e)}")
            raise e
        finally:
            stmt.close()
            conn.close()

class TargetFactory:
    _handlers = {
        'FILE': FileTargetHandler,
        'DB': DBTargetHandler
    }
    
    @classmethod
    def get_handler(cls, target_type: str) -> TargetHandler:
        target_type = str(target_type).upper()
        if target_type not in cls._handlers:
             raise ValueError(f"Unsupported target type: {target_type}")
        return cls._handlers[target_type]()
