import React, { useState } from 'react';
import { TransformationConfig, TransformationSpec } from '../types';
import apiClient, { ApiError } from '../services/apiClient';

interface JsonPreviewProps {
  config: TransformationConfig;
  onSpecChange?: (spec: TransformationSpec) => void;
}

const JsonPreview: React.FC<JsonPreviewProps> = ({ config, onSpecChange }) => {
  const [, setJsonSpec] = useState<TransformationSpec | null>(null);
  const [jsonString, setJsonString] = useState<string>('');
  const [isEditing, setIsEditing] = useState(false);
  const [isLoading, setIsLoading] = useState(false);
  const [error, setError] = useState<string | null>(null);

  const generateJsonFromAPI = async () => {
    setIsLoading(true);
    setError(null);
    
    try {
      const spec = await apiClient.generateJson(config);
      setJsonSpec(spec);
      setJsonString(JSON.stringify(spec, null, 2));
      
      if (onSpecChange) {
        onSpecChange(spec);
      }
    } catch (err) {
      if (err instanceof ApiError) {
        setError(`API Error: ${err.message}`);
      } else {
        setError(`Error: ${err instanceof Error ? err.message : 'Unknown error occurred'}`);
      }
    } finally {
      setIsLoading(false);
    }
  };

  return (
    <div className="json-preview">
      <div className="json-preview-header">
        <h3>JSON Specification Preview</h3>
        <div className="json-preview-controls">
          <button
            onClick={generateJsonFromAPI}
            className="btn btn-primary"
            disabled={isLoading}
          >
            {isLoading ? 'Generating...' : 'Generate JSON'}
          </button>
          <button
            onClick={() => setIsEditing(!isEditing)}
            className="btn btn-secondary"
            disabled={!jsonString}
          >
            {isEditing ? 'View Mode' : 'Edit Mode'}
          </button>
        </div>
      </div>

      <div style={{ padding: '20px', backgroundColor: '#f8f9fa', borderRadius: '4px', margin: '20px 0' }}>
        <h4>Configuration Summary:</h4>
        <p><strong>Source:</strong> {config.dataSource.source?.type || 'Not configured'}</p>
        <p><strong>Target:</strong> {config.dataSource.target?.type || 'Not configured'}</p>
        <p><strong>Field Transformations:</strong> {config.fieldTransforms.length}</p>
        <p><strong>Row Transformations:</strong> {config.rowTransforms.length}</p>
      </div>

      {error && (
        <div style={{ 
          padding: '15px', 
          backgroundColor: '#f8d7da', 
          border: '1px solid #f5c6cb', 
          borderRadius: '4px', 
          margin: '20px 0',
          color: '#721c24'
        }}>
          <h4>Error</h4>
          <p>{error}</p>
        </div>
      )}

      {isLoading && (
        <div style={{ 
          padding: '20px', 
          textAlign: 'center', 
          backgroundColor: '#f8f9fa', 
          borderRadius: '4px', 
          margin: '20px 0' 
        }}>
          <div style={{ 
            border: '4px solid #f3f3f3', 
            borderTop: '4px solid #007bff', 
            borderRadius: '50%', 
            width: '40px', 
            height: '40px', 
            animation: 'spin 1s linear infinite', 
            margin: '0 auto 15px' 
          }}></div>
          <p>Generating JSON specification...</p>
        </div>
      )}

      {jsonString && (
        <div className="json-editor-container">
          <div className="json-editor-header">
            <h4>Generated JSON Specification</h4>
          </div>
          
          <div className="json-editor">
            <textarea
              style={{
                width: '100%',
                height: '400px',
                fontFamily: 'monospace',
                fontSize: '12px',
                padding: '10px',
                border: '1px solid #ccc',
                borderRadius: '4px',
                backgroundColor: '#f8f9fa'
              }}
              value={jsonString}
              onChange={isEditing ? (e) => {
                setJsonString(e.target.value);
                try {
                  const parsed = JSON.parse(e.target.value);
                  setJsonSpec(parsed);
                  if (onSpecChange) onSpecChange(parsed);
                } catch (err) {
                  // Invalid JSON, but don't break the UI
                }
              } : undefined}
              readOnly={!isEditing}
            />
          </div>
        </div>
      )}

      <div style={{ padding: '20px', backgroundColor: '#e7f3ff', borderRadius: '4px', margin: '20px 0' }}>
        <h4>Next Steps:</h4>
        <ul>
          <li>Click "Generate JSON" to create the transformation specification</li>
          <li>Review the generated JSON</li>
          <li>Edit if needed using "Edit Mode"</li>
          <li>Proceed to Code Generation step</li>
        </ul>
      </div>
    </div>
  );
};

export default JsonPreview;
