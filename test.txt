from pyspark.sql import SparkSession
from datetime import datetime
import logging
from typing import Dict

class AuditMetrics:
    """Captures and tracks audit metrics for a PySpark ETL run."""
    
    def __init__(self, config: Dict):
        self.config = config.get("audit", {})
        self.enabled = self.config.get("enable", False)
        self.capture_row_count = self.config.get("capture_row_count", True)
        self.capture_duration = self.config.get("capture_duration", True)
        self.audit_path = self.config.get("path")
        
        self.metrics = {
            'pipeline_id': None,
            'start_time': None,
            'end_time': None,
            'duration_seconds': None,
            'source_row_count': 0,
            'target_row_count': 0,
            'status': 'INITIALIZED'
        }
        
    def start_pipeline(self, pipeline_id: str):
        """Marks the beginning of the pipeline run."""
        if not self.enabled:
            return
            
        self.metrics['pipeline_id'] = pipeline_id
        if self.capture_duration:
            self.metrics['start_time'] = datetime.now()
        self.metrics['status'] = 'RUNNING'
        logging.info(f"Audit: Started pipeline {pipeline_id} at {self.metrics['start_time']}")

    def record_source_count(self, count: int):
        """Records the number of rows read from the source."""
        if self.enabled and self.capture_row_count:
            self.metrics['source_row_count'] = count
            logging.info(f"Audit: Recorded Source Count: {count}")

    def record_target_count(self, count: int):
        """Records the number of rows written to the target."""
        if self.enabled and self.capture_row_count:
            self.metrics['target_row_count'] = count
            logging.info(f"Audit: Recorded Target Count: {count}")

    def end_pipeline(self, spark: SparkSession, status: str = 'SUCCESS', error_msg: str = None):
        """
        Marks the end of the pipeline run, calculates duration, 
        and writes the audit record to the configured path.
        """
        if not self.enabled:
            return

        self.metrics['status'] = status
        
        if self.capture_duration and self.metrics['start_time']:
            self.metrics['end_time'] = datetime.now()
            duration = self.metrics['end_time'] - self.metrics['start_time']
            self.metrics['duration_seconds'] = duration.total_seconds()
            
        if error_msg:
            self.metrics['error_message'] = str(error_msg)

        logging.info(f"Audit: Pipeline Completed with Status: {status}. Total Time: {self.metrics.get('duration_seconds')}s")

        # Write audit log to destination
        self._write_audit_log(spark)

    def _write_audit_log(self, spark: SparkSession):
        """Internal method to write the metrics as a JSON file using native Python I/O.
        
        Previously used PySpark DataFrame writer (audit_df.write.json) which suffered
        from a race condition: Spark's lazy/async write produced empty part-*.json files
        that were moved before content was flushed to disk. Now uses json.dump for
        reliable, atomic single-file writes.
        """
        if not self.audit_path:
            logging.warning("Audit is enabled, but no 'path' was provided in the JSON configuration. Skipping write.")
            return
            
        try:
            import os
            import json
            
            # Convert datetime objects to strings for JSON serialization
            metrics_out = dict(self.metrics)
            if metrics_out.get('start_time') and hasattr(metrics_out['start_time'], 'strftime'):
                metrics_out['start_time'] = metrics_out['start_time'].strftime("%Y-%m-%d %H:%M:%S")
            if metrics_out.get('end_time') and hasattr(metrics_out['end_time'], 'strftime'):
                metrics_out['end_time'] = metrics_out['end_time'].strftime("%Y-%m-%d %H:%M:%S")

            # Build destination path
            pipeline_name = self.metrics.get('pipeline_id', 'unknown')
            timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
            os.makedirs(self.audit_path, exist_ok=True)
            final_audit_file = os.path.join(self.audit_path, f"audit_{pipeline_name}_{timestamp}.json")
            
            logging.info(f"Writing single-file Audit Log to: {final_audit_file}")
            
            # Write using native Python — reliable and atomic for a single metrics dict
            with open(final_audit_file, 'w') as f:
                json.dump(metrics_out, f, indent=2)
            
        except Exception as e:
            logging.error(f"Failed to write Audit Log to {self.audit_path}: {str(e)}")
