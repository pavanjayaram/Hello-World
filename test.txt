from pyspark.sql import DataFrame
from pyspark.sql import functions as F
import logging
from typing import Dict

class RejectHandler:
    """
    Handles capturing, formatting, and storing rows that fail 
    business validation rules (Filters, Deduplication, Null Constraints).
    """
    
    def __init__(self, config: Dict):
        # We look specifically for the "reject" block in the JSON
        self.config = config.get("reject", {})
        self.enabled = self.config.get("enable", False)
        self.path = self.config.get("path")
        self.include_reason = self.config.get("include_reason", True)
        
        # We also look for the "error_handling" block to determine failure behavior
        # Fallback to errorHandling for raw exports from UI
        error_config = config.get("error_handling", config.get("errorHandling", {}))
        self.error_mode = error_config.get("mode", "reject_and_continue")
        
        self.total_rejects = 0 
        
    def write_rejects(self, df: DataFrame, reason: str = 'VALIDATION_FAILED'):
        """
        Takes a DataFrame of "bad" rows, appends the failure reason, 
        and writes them out to the reject path.
        """
        if not self.enabled:
            return
            
        # Count how many bad rows we caught in this specific validation step
        count = df.count()
        if count == 0:
            return
            
        self.total_rejects += count
        
        # If enabled, append the reason and the timestamp so analysts can debug later
        if self.include_reason:
            df = df.withColumn('reject_reason', F.lit(reason))
            df = df.withColumn('reject_timestamp', F.current_timestamp())
            
        if not self.path:
            logging.warning("Rejects are enabled, but no 'path' is provided in JSON. Cannot write broken rows.")
            return
            
        logging.warning(f"Validation Error: Caught {count} rejected rows. Reason: {reason}.")
        logging.info(f"Writing rejected rows to {self.path}")
        
        try:
            # We typically append Parquet for rejects so we don't accidentally overwrite past failures
            df.write.mode("append").parquet(self.path)
            
            # If the user specifically set reject_and_fail in their JSON, we crash the pipeline immediately!
            if self.error_mode == "reject_and_fail":
                error_msg = f"Pipeline aborted. Error Mode is 'reject_and_fail' and {count} rows failed validation: {reason}"
                logging.error(error_msg)
                raise RuntimeError(error_msg)
                
        except Exception as e:
            logging.error(f"Failed to write rejected rows to {self.path}: {str(e)}")
            
    def get_reject_count(self) -> int:
        return self.total_rejects
