import json
import os
import logging
from pyspark.sql import SparkSession
from pyspark.sql.window import Window
from transformations import apply_all_transformations, apply_field_mappings
from audit_logger import AuditMetrics
from reject_handler import RejectHandler
from sources import SourceFactory
from targets import TargetFactory

class FileToFileETL:
    def __init__(self, config_path, config=None):
        """
        Initializes the ETL Engine.
        """
        self.config_path = config_path
        self.config = config if config else self._load_config()
        self.spark = self._create_spark_session()
        self.audit = AuditMetrics(self.config)
        self.reject = RejectHandler(self.config)
        
    def _load_config(self):
        logging.info(f"Loading configuration from: {self.config_path}")
        with open(self.config_path, 'r') as f:
            return json.load(f)
            
    def _create_spark_session(self):
        app_name = self.config.get("pipeline", {}).get("pipeline_id", "Dynamic_ETL")
        logging.info(f"Initializing SparkSession: {app_name}")
        
        import tempfile, glob, sys as _sys
        
        # Find JDBC driver JARs
        jars_dir = os.path.join(os.path.dirname(os.path.abspath(__file__)), 'jars')
        jar_files = glob.glob(os.path.join(jars_dir, '*.jar')) if os.path.isdir(jars_dir) else []
        
        # Java 17+ and Hadoop Windows workarounds
        os.environ.setdefault('JAVA_TOOL_OPTIONS', (
            "--add-opens=java.base/sun.nio.ch=ALL-UNNAMED "
            "--add-opens=java.base/java.lang=ALL-UNNAMED "
            "--add-opens=java.base/java.lang.reflect=ALL-UNNAMED "
            "--add-opens=java.base/java.io=ALL-UNNAMED "
            "--add-opens=java.base/java.util=ALL-UNNAMED "
            "--add-exports=java.base/sun.nio.ch=ALL-UNNAMED "
            "--add-exports=java.base/javax.security.auth=ALL-UNNAMED"
        ))
        os.environ['HADOOP_USER_NAME'] = 'spark'
        
        builder = SparkSession.builder \
            .appName(app_name) \
            .config("spark.hadoop.hadoop.native.lib", "false") \
            .config("spark.hadoop.io.nativeio.NativeIO.Windows.access0", "false") \
            .config("spark.hadoop.hadoop.security.authentication", "simple") \
            .config("spark.hadoop.hadoop.security.authorization", "false") \
            .config("spark.local.dir", tempfile.gettempdir())
        
        if jar_files:
            # Convert to file:/// URIs — raw backslash paths break PySpark on Windows
            def _to_uri(p): return 'file:///' + p.replace('\\', '/')
            jars_csv = ','.join([_to_uri(f) for f in jar_files])
            builder = builder \
                .config("spark.jars", jars_csv) \
                .config("spark.driver.extraClassPath", ','.join(jar_files).replace('\\', '/')) \
                .config("spark.executor.extraClassPath", ','.join(jar_files).replace('\\', '/'))
        
        spark = builder.getOrCreate()
            
        log_level = self.config.get("pipeline", {}).get("log_level", "WARN")
        spark.sparkContext.setLogLevel(log_level)
        return spark

    def extract(self):
        """
        Reads the source file dynamically based on the JSON configuration via SourceFactory.
        """
        source_config = self.config.get("source", {})
        source_type = source_config.get("type", "FILE")
        
        logging.info(f"Extracting data using SourceFactory for type: {source_type}")
        handler = SourceFactory.get_handler(source_type)
        df = handler.read(self.spark, source_config)
        
        # Capture source count
        row_count = df.count()
        logging.info(f"Extracted {row_count} rows.")
        self.audit.record_source_count(row_count)
        
        return df

    def transform(self, df):
        """
        Applies transformations and field mappings using the central transformation engine.
        """
        logging.info("Starting Transformations via Central Engine...")
        
        # 1. Apply rules from "transformations" array
        # apply_all_transformations handles row_number, if_then_else, filter, sorting, etc.
        # We pass self.reject in the context dictionary so transformations can log bad rows!
        context = {
            "reject_handler": self.reject
        }
        df = apply_all_transformations(self, df, self.config, context)
        
        # 2. Apply "field_mappings" (Selecting and renaming target columns in specified order)
        df = apply_field_mappings(df, self.config)
            
        return df

    def load(self, df):
        """
        Writes the transformed DataFrame to the target destination via TargetFactory.
        """
        target_config = self.config.get("target", {})
        target_type = target_config.get("type", "FILE")
        
        # Capture target count BEFORE writing (Action trigger)
        row_count = df.count()
        self.audit.record_target_count(row_count)
            
        logging.info(f"Loading data using TargetFactory for type: {target_type}")
        handler = TargetFactory.get_handler(target_type)
        handler.write(df, target_config)
        logging.info("Data load complete!")

    def run(self):
        """
        Orchestrates the ETL pipeline.
        """
        pipeline_id = self.config.get('pipeline', {}).get('pipeline_id', 'Unknown_Pipeline')
        logging.info(f"=== Starting ETL Pipeline: {pipeline_id} ===")
        
        try:
            # 1. Start Audit tracking
            self.audit.start_pipeline(pipeline_id)
            
            # 2. Extract
            source_df = self.extract()
            
            # 3. Transform (Rejects are captured automatically in this specific step based on context)
            transformed_df = self.transform(source_df)
            
            # 4. Load (Audit captures row_count and handles writes)
            self.load(transformed_df)
            
            # 5. Check if the pipeline finished, but generated rejections
            reject_count = self.reject.get_reject_count()
            if reject_count > 0:
                logging.warning(f"--- WARNING: {reject_count} invalid rows were sent to Reject path ---")
            
            # 6. Successfully terminate Auditing (including rejects in the final audit counts)
            self.audit.end_pipeline(self.spark, status='SUCCESS')
            logging.info("=== ETL Pipeline Execution Finished Successfully ===")
            
        except Exception as e:
            # Mark the Audit log as FAILED with the error reason
            self.audit.end_pipeline(self.spark, status='FAILED', error_msg=str(e))
            logging.error(f"=== ETL Pipeline FAILED: {str(e)} ===")
            raise e

if __name__ == "__main__":
    import sys
    
    if len(sys.argv) < 2:
        print("ERROR: Config file path is required.", file=sys.stderr)
        print("Usage: python file_to_file_etl.py <config.json>", file=sys.stderr)
        sys.exit(2)
    
    config_file = sys.argv[1]
    engine = FileToFileETL(config_file)
    engine.run()
