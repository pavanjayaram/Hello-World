import React, { useState, useEffect } from 'react';
import apiClient from '../services/apiClient';
import './FileBrowserModal.css';

interface FileEntry {
    name: string;
    path: string;
    is_dir: boolean;
    size: number;
    modified: string;
}

interface FileBrowserModalProps {
    isOpen: boolean;
    onClose: () => void;
    onSelect: (path: string) => void;
    initialPath?: string;
}

const FileBrowserModal: React.FC<FileBrowserModalProps> = ({
    isOpen,
    onClose,
    onSelect,
    initialPath
}) => {
    const [currentPath, setCurrentPath] = useState<string>('');
    const [parentPath, setParentPath] = useState<string | null>(null);
    const [files, setFiles] = useState<FileEntry[]>([]);
    const [loading, setLoading] = useState(false);
    const [error, setError] = useState<string | null>(null);
    const [selectedFile, setSelectedFile] = useState<string | null>(null);

    // Load files when modal opens or path changes
    useEffect(() => {
        if (isOpen) {
            loadDirectory(currentPath || initialPath);
        }
    }, [isOpen]);

    const loadDirectory = async (path?: string) => {
        setLoading(true);
        setError(null);
        try {
            const result = await apiClient.listDirectory(path);
            setCurrentPath(result.current_path);
            setParentPath(result.parent_path);
            setFiles(result.items);
        } catch (err: any) {
            console.error("Failed to list directory:", err);
            setError(err.message || "Failed to load directory");
        } finally {
            setLoading(false);
        }
    };

    const handleEntryClick = (entry: FileEntry) => {
        if (entry.is_dir) {
            loadDirectory(entry.path);
        } else {
            setSelectedFile(entry.path);
        }
    };

    const handleSelect = () => {
        if (selectedFile) {
            onSelect(selectedFile);
            onClose();
        }
    };

    const formatSize = (bytes: number) => {
        if (bytes === 0) return '';
        const sizes = ['B', 'KB', 'MB', 'GB', 'TB'];
        const i = Math.floor(Math.log(bytes) / Math.log(1024));
        return `${(bytes / Math.pow(1024, i)).toFixed(1)} ${sizes[i]}`;
    };

    if (!isOpen) return null;

    return (
        <div className="file-browser-overlay">
            <div className="file-browser-modal">
                <div className="file-browser-header">
                    <h3>Browse Server Files</h3>
                    <button className="close-btn" onClick={onClose}>&times;</button>
                </div>

                <div className="file-browser-path">
                    <button
                        className="nav-btn"
                        onClick={() => parentPath && loadDirectory(parentPath)}
                        disabled={!parentPath}
                        title="Go Up"
                    >
                        ⬆️ Up
                    </button>
                    <input
                        type="text"
                        value={currentPath}
                        readOnly
                        className="path-input"
                    />
                </div>

                <div className="file-browser-content">
                    {loading ? (
                        <div className="loading">Loading...</div>
                    ) : error ? (
                        <div className="error">{error}</div>
                    ) : (
                        <div className="file-list">
                            {parentPath && (
                                <div
                                    className="file-entry directory"
                                    onClick={() => loadDirectory(parentPath)}
                                >
                                    <span className="icon">📁</span>
                                    <span className="name">..</span>
                                </div>
                            )}
                            {files.map((file) => (
                                <div
                                    key={file.path}
                                    className={`file-entry ${file.is_dir ? 'directory' : 'file'} ${selectedFile === file.path ? 'selected' : ''}`}
                                    onClick={() => handleEntryClick(file)}
                                    onDoubleClick={() => {
                                        if (file.is_dir) loadDirectory(file.path);
                                        else {
                                            setSelectedFile(file.path);
                                            handleSelect();
                                        }
                                    }}
                                >
                                    <span className="icon">{file.is_dir ? '📁' : '📄'}</span>
                                    <span className="name">{file.name}</span>
                                    <span className="size">{formatSize(file.size)}</span>
                                    <span className="date">{new Date(file.modified).toLocaleDateString()}</span>
                                </div>
                            ))}
                        </div>
                    )}
                </div>

                <div className="file-browser-footer">
                    <div className="selected-info">
                        {selectedFile ? `Selected: ${selectedFile.split(/[/\\]/).pop()}` : 'No file selected'}
                    </div>
                    <div className="actions">
                        <button className="btn btn-secondary" onClick={onClose}>Cancel</button>
                        <button
                            className="btn btn-primary"
                            onClick={handleSelect}
                            disabled={!selectedFile}
                        >
                            Select File
                        </button>
                    </div>
                </div>
            </div>
        </div>
    );
};

export default FileBrowserModal;
