def check_api(cfg):
    name = cfg.get("name", "UNKNOWN")
    url = cfg.get("url")
    enabled = cfg.get("enabled", True)
    ignore_if_down = cfg.get("ignore_if_down", True)

    if not enabled:
        return True, f"{name} -> SKIPPED (disabled)"

    log(f"Checking {name}...")

    ATTEMPTS = 3
    SLOW_THRESHOLD = 5

    for attempt in range(1, ATTEMPTS + 1):
        try:
            response = requests.get(
                url,
                timeout=TIMEOUT,
                verify=False
            )

            response_time = response.elapsed.total_seconds()

            log(f"{name} -> Attempt {attempt} | Time: {response_time:.2f}s")

            # ❌ HTTP error (real issue)
            if response.status_code != 200:
                return False, (
                    f"{name} -> FAILED ({response.status_code}) "
                    f"on attempt {attempt}"
                )

            # ❌ Slow response
            if response_time > SLOW_THRESHOLD:
                return False, (
                    f"{name} -> SLOW ({response_time:.2f}s)"
                )

        except requests.exceptions.Timeout:
            if ignore_if_down:
                return True, f"{name} -> DOWN (ignored)"
            return False, f"{name} -> TIMEOUT"

        except requests.exceptions.ConnectionError:
            if ignore_if_down:
                return True, f"{name} -> DOWN (ignored)"
            return False, f"{name} -> CONNECTION ERROR"

        except Exception as e:
            return False, f"{name} -> ERROR ({str(e)})"

    return True, f"{name} -> HEALTHY"
