import requests
import os

class OctaneClient:

    def __init__(self):
        self.base_url = os.getenv("OCTANE_URL")
        self.client_id = os.getenv("OCTANE_CLIENT_ID")
        self.client_secret = os.getenv("OCTANE_CLIENT_SECRET")
        self.space_id = os.getenv("OCTANE_SPACE_ID")
        self.workspace_id = os.getenv("OCTANE_WORKSPACE_ID")
        self.cookies = None

    def authenticate(self):
        res = requests.post(
            f"{self.base_url}/authentication/sign_in",
            auth=(self.client_id, self.client_secret)
        )
        self.cookies = res.cookies

    def create_run(self, name):
        url = f"{self.base_url}/api/shared_spaces/{self.space_id}/workspaces/{self.workspace_id}/runs"
        payload = {"data": [{"name": name, "status": {"phase": "Running"}}]}
        res = requests.post(url, json=payload, cookies=self.cookies)
        return res.json()["data"][0]["id"]

    def update_run(self, run_id, status):
        url = f"{self.base_url}/api/shared_spaces/{self.space_id}/workspaces/{self.workspace_id}/runs/{run_id}"
        payload = {"data": [{"id": run_id, "status": {"phase": status}}]}
        requests.put(url, json=payload, cookies=self.cookies)

    def attach_report(self, file_path):
        url = f"{self.base_url}/api/shared_spaces/{self.space_id}/workspaces/{self.workspace_id}/attachments"
        files = {"file": open(file_path, "rb")}
        requests.post(url, files=files, cookies=self.cookies)
