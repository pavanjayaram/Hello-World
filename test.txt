def check_api(cfg):
    name = cfg.get("name", "UNKNOWN")
    method = cfg.get("method", "GET").upper()

    env_config = cfg.get("environments", {}).get(ENV)

    if not env_config:
        return False, f"{name} -> No config for ENV {ENV}"

    url = env_config["url"]

    log(f"[{ENV}] Checking {name}...")

    ATTEMPTS = 5
    SLOW_THRESHOLD = 5  # seconds

    for attempt in range(1, ATTEMPTS + 1):
        try:
            response = requests.get(
                url,
                timeout=TIMEOUT,
                verify=False
            )

            response_time = response.elapsed.total_seconds()

            # 🔍 Capture node info from headers
            headers = response.headers
            node_info = (
                headers.get("X-Node-IP") or
                headers.get("X-Backend-Server") or
                headers.get("X-Served-By") or
                headers.get("Server") or
                "UNKNOWN"
            )

            log(f"{name} -> Attempt {attempt} | Node: {node_info} | Time: {response_time:.2f}s")

            # ❌ HTTP failure
            if response.status_code != 200:
                return False, (
                    f"{name} -> FAILED (Status {response.status_code}) "
                    f"on attempt {attempt} | Node: {node_info}"
                )

            # ❌ Slow response
            if response_time > SLOW_THRESHOLD:
                return False, (
                    f"{name} -> SLOW ({response_time:.2f}s) "
                    f"on attempt {attempt} | Node: {node_info}"
                )

        except requests.exceptions.Timeout:
            return False, f"{name} -> TIMEOUT on attempt {attempt}"

        except requests.exceptions.ConnectionError:
            return False, f"{name} -> CONNECTION ERROR on attempt {attempt}"

        except Exception as e:
            return False, f"{name} -> ERROR ({str(e)}) on attempt {attempt}"

    return True, f"{name} -> HEALTHY (All {ATTEMPTS} checks passed)"
