# save as test_spark.py in backend folder, run: python test_spark.py
import os, tempfile
os.environ.setdefault('JAVA_TOOL_OPTIONS', '--add-opens=java.base/sun.nio.ch=ALL-UNNAMED --add-opens=java.base/java.lang=ALL-UNNAMED')
os.environ['HADOOP_USER_NAME'] = 'spark'

from pyspark.sql import SparkSession
spark = SparkSession.builder \
    .appName("test") \
    .config("spark.hadoop.hadoop.native.lib", "false") \
    .config("spark.hadoop.io.nativeio.NativeIO.Windows.access0", "false") \
    .config("spark.hadoop.hadoop.security.authentication", "simple") \
    .config("spark.hadoop.hadoop.security.authorization", "false") \
    .config("spark.local.dir", tempfile.gettempdir()) \
    .getOrCreate()
print("SUCCESS!")
spark.stop()
