import json
from src.api_client import call_api
from src.xml_parser import xml_to_dict
from src.comparator import compare
from src.report_generator import generate_csv_report, generate_html_report
from src.octane_client import OctaneClient

def main():
    with open("config/config.json") as f:
        config = json.load(f)

    octane = OctaneClient()
    octane.authenticate()

    for test in config["test_cases"]:
        print(f"Running: {test['test_name']}")

        run_id = octane.create_run(test["test_name"])

        dev_res = call_api(test["dev_url"], test["request_payload"], test["soap_action"])
        uat_res = call_api(test["uat_url"], test["request_payload"], test["soap_action"])

        if "ERROR" in dev_res or "ERROR" in uat_res:
            octane.update_run(run_id, "Failed")
            continue

        dev_data = xml_to_dict(dev_res)
        uat_data = xml_to_dict(uat_res)

        mismatches, match_percent = compare(
            dev_data,
            uat_data,
            test["fields_to_compare"],
            test["ignore_fields"]
        )

        status = "Passed" if not mismatches else "Failed"

        csv_report = generate_csv_report(test["test_name"], mismatches)
        html_report = generate_html_report(test["test_name"], mismatches, match_percent)

        octane.update_run(run_id, status)
        octane.attach_report(run_id, html_report)

        print(f"{test['test_name']} → {status} ({match_percent:.2f}%)")

if __name__ == "__main__":
    main()
