def normalize_value(val):
    """
    Normalize values for fair comparison
    - Trim strings
    - Convert numeric strings to numbers
    """
    if val is None:
        return None

    if isinstance(val, str):
        val = val.strip()

        # Try numeric conversion
        if val.isdigit():
            return int(val)

        try:
            return float(val)
        except:
            return val

    return val


def is_ignored(path, ignore_fields):
    """
    Check if current path should be ignored
    Supports partial match
    """
    return any(ig in path for ig in ignore_fields)


def compare_values(dev, uat, path):
    """
    Compare leaf values
    """
    dev_val = normalize_value(dev)
    uat_val = normalize_value(uat)

    if dev_val != uat_val:
        return [{
            "field": path,
            "dev": dev_val,
            "uat": uat_val,
            "type": "VALUE_MISMATCH"
        }]
    return []


def compare_lists(dev_list, uat_list, path, ignore_fields):
    """
    Compare lists (order-sensitive)
    """
    mismatches = []

    max_len = max(len(dev_list), len(uat_list))

    for i in range(max_len):
        new_path = f"{path}[{i}]"

        if i >= len(dev_list):
            mismatches.append({
                "field": new_path,
                "dev": None,
                "uat": uat_list[i],
                "type": "MISSING_IN_DEV"
            })
        elif i >= len(uat_list):
            mismatches.append({
                "field": new_path,
                "dev": dev_list[i],
                "uat": None,
                "type": "MISSING_IN_UAT"
            })
        else:
            mismatches.extend(
                deep_compare(dev_list[i], uat_list[i], new_path, ignore_fields)
            )

    return mismatches


def deep_compare(dev, uat, path="", ignore_fields=None):
    """
    Main recursive comparison function
    """
    if ignore_fields is None:
        ignore_fields = []

    mismatches = []

    # Ignore logic
    if is_ignored(path, ignore_fields):
        return mismatches

    # Type mismatch
    if type(dev) != type(uat):
        return [{
            "field": path,
            "dev": dev,
            "uat": uat,
            "type": "TYPE_MISMATCH"
        }]

    # Dict comparison
    if isinstance(dev, dict):
        all_keys = set(dev.keys()).union(set(uat.keys()))

        for key in all_keys:
            new_path = f"{path}.{key}" if path else key

            if key not in dev:
                mismatches.append({
                    "field": new_path,
                    "dev": None,
                    "uat": uat[key],
                    "type": "MISSING_IN_DEV"
                })
            elif key not in uat:
                mismatches.append({
                    "field": new_path,
                    "dev": dev[key],
                    "uat": None,
                    "type": "MISSING_IN_UAT"
                })
            else:
                mismatches.extend(
                    deep_compare(dev[key], uat[key], new_path, ignore_fields)
                )

    # List comparison
    elif isinstance(dev, list):
        mismatches.extend(compare_lists(dev, uat, path, ignore_fields))

    # Leaf node comparison
    else:
        mismatches.extend(compare_values(dev, uat, path))

    return mismatches
