import json
import os
import logging
from pyspark.sql import SparkSession
from transformations import apply_all_transformations, apply_field_mappings
from audit_logger import AuditMetrics
from reject_handler import RejectHandler
from sources import SourceFactory
from targets import TargetFactory

class FileToDBETL:
    def __init__(self, config_path, config=None):
        """
        Initializes the ETL Engine for File/DB to DB.
        """
        self.config_path = config_path
        self.config = config if config else self._load_config()
        self.spark = self._create_spark_session()
        self.audit = AuditMetrics(self.config)
        self.reject = RejectHandler(self.config)
        
    def _load_config(self):
        logging.info(f"Loading configuration from: {self.config_path}")
        with open(self.config_path, 'r') as f:
            return json.load(f)
            
    def _create_spark_session(self):
        app_name = self.config.get("pipeline", {}).get("pipeline_id", "Dynamic_ETL_F2DB")
        logging.info(f"Initializing SparkSession: {app_name}")
        builder = SparkSession.builder.appName(app_name)
        
        # Look for custom JARs (like excel reader or DB drivers)
        jars = self.config.get("pipeline", {}).get("jars", [])
        if jars:
            maven_jars = [j for j in jars if ":" in j]
            local_jars = [j for j in jars if ":" not in j]
            
            if maven_jars:
                builder = builder.config("spark.jars.packages", ",".join(maven_jars))
            if local_jars:
                jar_str = ",".join(local_jars)
                # spark.jars is always comma-separated
                builder = builder.config("spark.jars", jar_str)
                # spark.driver.extraClassPath requires OS-specific separator (colon on Mac, semicolon on Windows)
                builder = builder.config("spark.driver.extraClassPath", os.pathsep.join(local_jars))
            
        spark = builder.getOrCreate()
        log_level = self.config.get("pipeline", {}).get("log_level", "WARN")
        spark.sparkContext.setLogLevel(log_level)
        return spark

    def extract(self):
        """
        Reads the source data dynamically based on the JSON configuration via SourceFactory.
        """
        source_config = self.config.get("source", {})
        source_type = source_config.get("type", "FILE")
        
        logging.info(f"Extracting data using SourceFactory for type: {source_type}")
        handler = SourceFactory.get_handler(source_type)
        df = handler.read(self.spark, source_config)
        
        row_count = df.count()
        logging.info(f"Extracted {row_count} rows.")
        self.audit.record_source_count(row_count)
        
        return df

    def transform(self, df):
        """
        Applies transformations and field mappings using the central transformation engine.
        """
        logging.info("Starting Transformations via Central Engine (CDC/Lookup enabled due to DB target)...")
        
        context = {
            "reject_handler": self.reject
        }
        df = apply_all_transformations(self, df, self.config, context)
        df = apply_field_mappings(df, self.config)
        return df

    def load(self, df):
        target_config = self.config.get("target", {})
        target_type = target_config.get("type", "DB")
        mode = target_config.get("mode", "append")
        
        row_count = df.count()
        self.audit.record_target_count(row_count)
        
        # Check if chunked load is enabled (APPEND mode only)
        chunked_config = self.config.get("chunked_load", {})
        if chunked_config.get("enabled", False) and mode == "append":
            logging.info(f"[FileToDBETL] Chunked load enabled — delegating to ChunkProcessor")
            from chunk_processor import ChunkProcessor
            processor = ChunkProcessor(self.config, self.spark)
            summary = processor.process_all_chunks(df)
            
            if summary.get("failed", 0) > 0:
                raise RuntimeError(
                    f"Chunked load completed with {summary['failed']} failed chunks. "
                    f"Use run_mode=RETRY_FAILED to retry. job_run_id={summary.get('job_run_id')}"
                )
            return
        
        logging.info(f"Loading data using TargetFactory for type: {target_type}")
        handler = TargetFactory.get_handler(target_type)
        # We pass the full self.config rather than just target_config so targets.py can scan for CDC
        handler.write(df, self.config)
        logging.info("Data load complete!")

    def run(self):
        pipeline_id = self.config.get('pipeline', {}).get('pipeline_id', 'Unknown_Pipeline')
        run_mode = self.config.get('pipeline', {}).get('run_mode', 'FULL').upper()
        logging.info(f"=== Starting FILE_TO_DB Pipeline: {pipeline_id} (run_mode: {run_mode}) ===")
        
        try:
            self.audit.start_pipeline(pipeline_id)
            
            if run_mode == 'RETRY_FAILED':
                # Retry mode: re-extract and re-transform, but only write failed chunks
                logging.info("[FileToDBETL] RETRY_FAILED mode — will reprocess only failed chunks")
                source_df = self.extract()
                transformed_df = self.transform(source_df)
                
                from chunk_processor import ChunkProcessor
                processor = ChunkProcessor(self.config, self.spark)
                job_run_id = self.config.get('chunked_load', {}).get('retry_job_run_id', None)
                summary = processor.retry_failed_chunks(transformed_df, job_run_id)
                
                if summary.get("failed", 0) > 0:
                    raise RuntimeError(
                        f"Retry completed with {summary['failed']} chunks still failing."
                    )
            else:
                # Normal FULL run
                source_df = self.extract()
                transformed_df = self.transform(source_df)
                self.load(transformed_df)
            
            reject_count = self.reject.get_reject_count()
            if reject_count > 0:
                logging.warning(f"--- WARNING: {reject_count} invalid rows were sent to Reject path ---")
                
            self.audit.end_pipeline(self.spark, status='SUCCESS')
            logging.info("=== FILE_TO_DB Pipeline Finished Successfully ===")
        except Exception as e:
            self.audit.end_pipeline(self.spark, status='FAILED', error_msg=str(e))
            logging.error(f"=== ETL Pipeline FAILED: {str(e)} ===")
            raise e
