def compare(dev_data, uat_data, fields, ignore_fields):
    mismatches = []
    total = 0
    matched = 0

    for field in fields:
        if field in ignore_fields:
            continue

        dev_val = dev_data.get(field)
        uat_val = uat_data.get(field)

        total += 1

        if dev_val == uat_val:
            matched += 1
        else:
            mismatches.append({
                "field": field,
                "dev": dev_val,
                "uat": uat_val
            })

    match_percent = (matched / total * 100) if total > 0 else 0

    return mismatches, match_percent
