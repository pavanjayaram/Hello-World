CDC_DEBUG_LOGS = []
def log_cdc(msg):
    timestamp = datetime.now().strftime("%H:%M:%S")
    formatted_msg = f"[{timestamp}] {msg}"
    CDC_DEBUG_LOGS.append(formatted_msg)

class TransformationHandler(ABC):
    @abstractmethod
    def apply(self, df: DataFrame, config: Dict, context: Dict = None) -> DataFrame:
        pass

class ColumnTransformationHandler(TransformationHandler):
    def apply(self, df: DataFrame, config: Dict, context: Dict = None) -> DataFrame:
        action = config.get('action')
        column = config.get('column')
        
        if action not in ['sequence', 'literal', 'current_date'] and (not column or column not in df.columns):
            raise ValueError(f"Invalid column: {column}")
        
        logging.info(f"Applying column transformation: {action} on {column}")
        
        if action == 'upper':
            return df.withColumn(column, F.upper(F.col(column)))
        elif action == 'lower':
            return df.withColumn(column, F.lower(F.col(column)))
        elif action == 'trim':
            return df.withColumn(column, F.trim(F.col(column)))
        elif action == 'cast':
            target_type = config.get('target_type')
            formats = config.get('formats')
            fmt = config.get('format')
            if formats and isinstance(formats, list) and target_type in ['timestamp', 'date', 'TimestampType', 'DateType']:
                func_name = "try_to_timestamp" if target_type in ['timestamp', 'TimestampType'] else "try_to_date"
                cols = [F.expr(f"{func_name}({column}, '{f}')") for f in formats]
                return df.withColumn(column, F.coalesce(*cols))
            elif fmt and target_type in ['timestamp', 'date', 'TimestampType', 'DateType']:
                func = F.to_timestamp if target_type in ['timestamp', 'TimestampType'] else F.to_date
                return df.withColumn(column, func(F.col(column), fmt))
            else:
                if target_type.lower() == 'date':
                    # Natively strip time components generated by driver/JVM default server TimeZones
                    return df.withColumn(column, F.date_format(F.to_date(F.col(column)), "yyyy-MM-dd"))
                return df.withColumn(column, F.col(column).cast(target_type))
        elif action == 'rename':
            return df.withColumnRenamed(column, config.get('new_name'))
        elif action == 'add':
            return df.withColumn(column, F.expr(config.get('expression')))
        elif action == 'drop':
            return df.drop(column)
        elif action == 'sequence':
            # High Performance Distributed ID Generator (Informatica equivalent for Big Data)
            # Generates unique, increasing 64-bit integers cleanly across hundreds of partitions.
            # E.g., Worker 1: 0,1,2 | Worker 2: 8589934592,8589934593
            return df.withColumn(column, F.monotonically_increasing_id())
        elif action == 'literal':
            val = config.get('value', '')
            dtype = config.get('data_type', 'string')
            
            # Handle empty values as None (Null) for non-string types to prevent cast errors
            if val == '' and dtype != 'string':
                lit_val = F.lit(None).cast(dtype if dtype != 'integer' else 'int')
            elif dtype == 'integer': 
                lit_val = F.lit(int(val) if val else 0)
            elif dtype in ['double', 'float']: 
                lit_val = F.lit(float(val) if val else 0.0)
            elif dtype == 'boolean': 
                lit_val = F.lit(str(val).lower() in ['true', '1'])
            elif dtype == 'date': 
                lit_val = F.to_date(F.lit(val))
            elif dtype == 'timestamp': 
                lit_val = F.to_timestamp(F.lit(val))
            else: 
                lit_val = F.lit(str(val))
            return df.withColumn(column, lit_val)
        elif action == 'current_date':
            return df.withColumn(column, F.current_date())
        elif action == 'if_then_else':
            return df.withColumn(column, F.when(F.expr(config.get('condition')), F.expr(config.get('trueValue'))).otherwise(F.expr(config.get('falseValue'))))
        elif action == 'data_masking':
            mask_type = config.get('masking_type', 'full')
            if mask_type == 'full': return df.withColumn(column, F.lit('****'))
            elif mask_type == 'hash': return df.withColumn(column, F.md5(F.col(column).cast('string')))
            elif mask_type == 'null': return df.withColumn(column, F.lit(None).cast('string'))
            elif mask_type == 'partial':
                sf = config.get('show_first', 2)
                sl = config.get('show_last', 2)
                sc = F.col(column).cast('string')
                ln = F.length(sc)
                return df.withColumn(column, F.when(F.col(column).isNull(), F.lit(None).cast('string')).when(ln <= (sf + sl), F.lit('****')).otherwise(F.concat(F.substring(sc, 1, sf), F.lit('****'), F.substring(sc, ln - sl + 1, sl))))
            elif mask_type == 'base64': return df.withColumn(column, F.base64(F.encode(F.col(column).cast('string'), 'utf-8')))
            elif mask_type == 'aes':
                key = config.get('key', '')
                return df.withColumn(column, F.expr(f"base64(aes_encrypt(cast({column} as string), '{key}'))"))
            return df
        elif action == 'data_unmasking':
            unmask_type = config.get('unmasking_type', 'base64')
            if unmask_type == 'base64': return df.withColumn(column, F.decode(F.unbase64(F.col(column)), 'utf-8'))
            elif unmask_type == 'aes':
                key = config.get('key', '')
                return df.withColumn(column, F.expr(f"cast(aes_decrypt(unbase64({column}), '{key}') as string)"))
            return df
        else:
            raise ValueError(f"Unsupported column action: {action}")

class RowTransformationHandler(TransformationHandler):
    def apply(self, df: DataFrame, config: Dict, context: Dict = None) -> DataFrame:
        action = config.get('action')
        logging.info(f"Applying row transformation: {action}")
        
        if action == 'filter':
            condition = config.get('condition')
            capture_rejects = config.get('capture_rejects', False)
            if capture_rejects and context and 'reject_handler' in context:
                rj = context['reject_handler']
                rejects = df.filter(f"NOT ({condition})")
                rj.write_rejects(rejects, config.get('reject_reason', f"Failed filter: {condition}"))
            return df.filter(condition)
        elif action == 'if':
            return df.withColumn(config.get('column'), F.when(F.expr(config.get('condition')), config.get('value')).otherwise(F.col(config.get('column'))))
        elif action in ['deduplicate', 'distinct']:
            raw_cols = config.get('columns', [])
            cols = [c.strip() for c in raw_cols.replace(';', ',').split(',')] if isinstance(raw_cols, str) else raw_cols
            handling = config.get('duplicate_handling', 'drop')
            
            if handling == 'fail':
                dup_counts = df.groupBy(cols).count().filter("count > 1")
                if dup_counts.count() > 0: raise ValueError(f"Duplicates found in {cols}")
                return df
            elif handling == 'capture':
                from pyspark.sql.window import Window
                part_cols = [F.col(c) for c in cols] if cols else [F.col(c) for c in df.columns]
                w = Window.partitionBy(*part_cols).orderBy(F.monotonically_increasing_id())
                df_w = df.withColumn("_row_num", F.row_number().over(w))
                if context and 'reject_handler' in context:
                    context['reject_handler'].write_rejects(df_w.filter("_row_num > 1").drop("_row_num"), "Duplicate record")
                return df_w.filter("_row_num == 1").drop("_row_num")
            else:
                return df.dropDuplicates(cols) if cols else df.dropDuplicates()
        elif action == 'sort':
            cols = config.get('columns', [])
            asc = config.get('ascending', True)
            sort_e = [F.col(c).asc() if asc else F.col(c).desc() for c in cols if c in df.columns]
            return df.orderBy(*sort_e) if sort_e else df
        elif action == 'sample':
            return df.sample(fraction=config.get('fraction', 0.1))
        else:
            raise ValueError(f"Unsupported row action: {action}")

class LookupTransformationHandler(TransformationHandler):
    def apply(self, df: DataFrame, config: Dict, context: Dict = None) -> DataFrame:
        lookup_table = config.get('lookup_table')
        operation = config.get('operation', 'join')
        
        if operation == 'surrogate_key':
            scol = config.get('source_column')
            tcol = config.get('target_column')
            try:
                try: lookup_df = DataFrameRegistry.get_dataframe(lookup_table)
                except:
                    stype = config.get('source_type')
                    path = config.get('path')
                    if stype and path:
                        sconf = {'type': stype, 'path': path, 'table': config.get('table') or path, 'schema': config.get('schema'), 'db_type': config.get('db_type'), 'connectionString': config.get('connection_string'), 'format': config.get('file_format', 'csv'), 'options': config.get('options', {})}
                        lookup_df = SourceFactory.get_handler(stype).read(df.sparkSession, sconf)
                        DataFrameRegistry.register(lookup_table, lookup_df)
                    else: raise ValueError(f"No source_type or path defined for lookup: {lookup_table}")
                max_val = lookup_df.agg(F.max(scol)).collect()[0][0]
                start_val = max_val if max_val is not None else 0
                logging.info(f"[Surrogate Key] MAX({scol}) from {lookup_table} = {max_val} → next value = {start_val + 1}")
            except Exception as e:
                logging.warning(f"[Surrogate Key] Failed to read MAX({scol}) from {lookup_table}: {e}. Defaulting to 1.")
                start_val = 0
            
            is_batch = any(x in tcol.upper() for x in ['LOAD_ID', 'BATCH_ID', 'JOB_ID'])
            if is_batch:
                logging.info(f"[Surrogate Key] Batch mode: setting {tcol} = {start_val + 1} for all rows")
                return df.withColumn(tcol, F.lit(start_val + 1))
            else:
                from pyspark.sql import Window
                w = Window.orderBy(F.monotonically_increasing_id())
                return df.withColumn(tcol, F.lit(start_val) + F.row_number().over(w))

        jkeys = config.get('join_keys', {})
        jtype = config.get('join_type', 'left')
        bcast = config.get('broadcast', False)
        
        try: lookup_df = DataFrameRegistry.get_dataframe(lookup_table)
        except:
            stype = config.get('source_type')
            path = config.get('path')
            sconf = {'type': stype, 'path': path, 'table': config.get('table') or path, 'schema': config.get('schema'), 'db_type': config.get('db_type'), 'connectionString': config.get('connection_string'), 'format': config.get('file_format', 'csv'), 'options': config.get('options', {})}
            lookup_df = SourceFactory.get_handler(stype).read(df.sparkSession, sconf)
            DataFrameRegistry.register(lookup_table, lookup_df)

        def resolve_col(val, frame):
            bare = val.split('.')[-1].strip('"').strip("'")
            return (frame[bare], bare, False) if bare in frame.columns else (F.lit(bare), bare, True)
            
        join_conds = []
        if isinstance(jkeys, dict):
            for s_raw, t_raw in jkeys.items():
                s_e, _, _ = resolve_col(s_raw, df)
                t_e, _, _ = resolve_col(t_raw, lookup_df)
                join_conds.append(s_e == t_e)
                
        final_cond = join_conds[0]
        for c in join_conds[1:]: final_cond = final_cond & c
        
        if bcast: lookup_df = F.broadcast(lookup_df)
        res = df.join(lookup_df, final_cond, jtype)
        return res

class CDCTransformationHandler(TransformationHandler):

    def apply(self, df: DataFrame, config: Dict, context: Dict = None) -> DataFrame:
        mode = config.get('mode')
        ttype = config.get('type')

        if mode == 'SCD_TYPE_2':
            df = df.withColumn(config.get('effective_start', 'eff_start_dt'), F.current_timestamp())
            df = df.withColumn(config.get('effective_end', 'eff_end_dt'), F.lit(None).cast('timestamp'))
            df = df.withColumn(config.get('current_flag', 'is_current'), F.lit(True))
            df = df.withColumn(config.get('operation_flag', 'op_type'), F.lit('INSERT'))
            return df

        elif ttype == 'cdc_compare' or mode == 'CDC_COMPARE':
            return self._apply_cdc_compare(df, config)

        return df


    def _normalize_column(self, col, col_type: str) -> F.Column:
        """Standardizes a column value for stable hashing while avoiding invalid casts on NULLs."""
        col_type = col_type.lower()
        c = F.col(col)
        
        # Check for Null/Empty first (to be used in final when)
        is_empty = (c.cast("string") == "") | (c.isNull())
        
        # 1. Date/Timestamp Normalization (Strip timezone offsets)
        if 'date' in col_type or 'timestamp' in col_type:
            ex_str = c.cast("string")
            cleaned_ex = F.regexp_replace(ex_str, r" [\+\-]\d{2}(:\d{2})?$", "")
            res = F.date_format(cleaned_ex.cast("timestamp"), "yyyy-MM-dd HH:mm:ss")
        # 2. Numeric Normalization (Use high-precision decimal)
        elif any(t in col_type for t in ['double', 'float', 'decimal', 'long', 'integer', 'int']):
            res = c.cast("decimal(20,2)").cast("string")
        # 3. String Normalization
        else:
            res = F.lower(F.trim(c.cast("string")))
            
        # Return "__NULL__" if originally empty or if normalization resulted in null.
        # This prevents casting the literal "__NULL__" to a numeric type during normalization.
        return F.when(is_empty | res.isNull(), F.lit("__NULL__")).otherwise(res)

    def _apply_cdc_compare(self, df: DataFrame, config: Dict) -> DataFrame:
        """
        CDC Compare — Optimized for large datasets (200K+).
        Instead of a full outer join (which materializes ALL rows including unchanged),
        uses 3 targeted joins that only produce changed rows:
          - LEFT ANTI  → Inserts (source keys not in target)
          - RIGHT ANTI → Deletes (target keys not in source)
          - INNER JOIN  → Updates (matching keys where checksum differs)
        """
        spark = df.sparkSession
        import time
        cdc_start = time.time()
        
        target_type = config.get('target_type', 'FILE')
        path = config.get('target_path')
        op_col = config.get('op_column', 'op_type')
        
        # --- 1. Resolve Key and Compare Columns ---
        key_cols = config.get('key_columns', [])
        s_kc = [k.strip() for k in config.get('source_key_columns', key_cols)]
        t_kc = [k.strip() for k in config.get('target_key_columns', key_cols)]
        
        compare_cols = config.get('compare_columns', [])
        s_cc = [c.strip() for c in config.get('source_compare_columns', compare_cols)]
        t_cc = [c.strip() for c in config.get('target_compare_columns', compare_cols)]
        
        exclude_cols = [e.strip() for e in config.get('exclude_columns', [])]
        
        if not s_kc:
            raise ValueError("Key columns required for CDC Compare")
        if not t_kc:
            t_kc = s_kc
        
        logging.info(f"[CDC] Compare against {target_type}: {path}")
        logging.info(f"[CDC] Key columns: source={s_kc}, target={t_kc}")
        
        # --- 2. Load Target Table ---
        sconf = {
            'type': target_type, 'path': path, 'table': path,
            'schema': config.get('schema'), 'db_type': config.get('db_type'),
            'connectionString': config.get('connection_string'),
            'format': config.get('format', 'parquet'),
            'options': config.get('options', {'header': 'true', 'inferSchema': 'true'})
        }
        
        target_is_empty = False
        try:
            target_df = SourceFactory.get_handler(target_type).read(spark, sconf)
            active_col = next((c for c in target_df.columns if c.lower() in ('active_flag', 'is_active', 'is_current')), None)
            if active_col:
                a_type = next((f.dataType.typeName().lower() for f in target_df.schema.fields if f.name == active_col), "string")
                if 'bool' in a_type:
                    target_df = target_df.filter(F.col(active_col) == True)
                elif any(t in a_type for t in ['int', 'long', 'short', 'byte']):
                    target_df = target_df.filter(F.col(active_col) == 1)
                else:
                    target_df = target_df.filter(F.col(active_col).isin('Y', 'y', '1'))
            
            target_df = target_df.cache()
            target_count = target_df.count()
            logging.info(f"[CDC] Target loaded: {target_count} active rows ({time.time() - cdc_start:.1f}s)")
            if target_count == 0:
                target_is_empty = True
        except Exception as e:
            logging.warning(f"[CDC] Failed to load target: {e}. All rows → INSERT.")
            target_df = spark.createDataFrame([], df.schema)
            target_is_empty = True

        # Fast path: empty target = all inserts
        if target_is_empty:
            logging.info("[CDC] Fast path: target empty — all INSERT")
            return df.withColumn(op_col, F.lit("I")) \
                     .withColumn("__debug_src", F.lit("")) \
                     .withColumn("__debug_tgt", F.lit(""))

        # --- 3. Determine Compare Columns ---
        if not s_cc:
            common = set(df.columns).intersection(set(target_df.columns))
            s_cc = sorted([c for c in common if c not in s_kc and c not in t_kc and c not in exclude_cols])
            t_cc = s_cc[:]
        if not t_cc:
            t_cc = s_cc[:]
        
        logging.info(f"[CDC] Checksum src cols: {s_cc}")
        logging.info(f"[CDC] Checksum tgt cols: {t_cc}")
        
        # --- 4. Build Normalization Helpers ---
        src_schema = {f.name: str(f.dataType).lower() for f in df.schema.fields}
        tgt_schema = {f.name: str(f.dataType).lower() for f in target_df.schema.fields}
        
        def norm_col(col_ref, col_name, schema):
            """Normalize a column for consistent checksum comparison."""
            dtype = schema.get(col_name, '')
            is_dt = 'date' in dtype or 'timestamp' in dtype
            is_num = any(x in dtype for x in ['double', 'float', 'decimal'])
            try:
                if is_dt:
                    s = col_ref.cast("string")
                    cleaned = F.regexp_replace(s, r" [\+\-]\d{2}(:\d{2})?$", "")
                    return F.date_format(cleaned.cast("timestamp"), "yyyy-MM-dd HH:mm:ss")
                elif is_num:
                    return col_ref.cast("decimal(20,2)").cast("string")
                else:
                    return col_ref.cast("string")
            except:
                return col_ref.cast("string")
        
        # --- 5. Pre-compute MD5 hash on source and target BEFORE joins ---
        # This is the key optimization: compute hash once, then use it in joins
        
        # Source: add __src_hash column
        src_hash_parts = [F.coalesce(norm_col(F.col(sc), sc, src_schema), F.lit("")) for sc in s_cc]
        src_hash_expr = F.md5(F.concat_ws("||", *src_hash_parts))
        df_with_hash = df.withColumn("__src_hash", src_hash_expr)
        
        # Target: select only key columns + hash (minimal projection for performance)
        tgt_hash_parts = [F.coalesce(norm_col(F.col(tc), tc, tgt_schema), F.lit("")) for tc in t_cc]
        tgt_hash_expr = F.md5(F.concat_ws("||", *tgt_hash_parts))
        # Keep all target columns for Delete row projection, plus the hash
        target_with_hash = target_df.withColumn("__tgt_hash", tgt_hash_expr)
        
        logging.info(f"[CDC] Hashes computed ({time.time() - cdc_start:.1f}s)")
        
        # --- 6. Three Targeted Joins (NO full outer join!) ---
        
        # Build simple key-based join condition (same column names on both sides)
        key_join = [df_with_hash[sk] == target_with_hash[tk] for sk, tk in zip(s_kc, t_kc)]
        key_cond = key_join[0]
        for c in key_join[1:]:
            key_cond = key_cond & c
        
        # 6a. INSERTS: source rows whose keys don't exist in target
        df_inserts = df_with_hash.join(F.broadcast(target_with_hash.select(t_kc)), key_cond, "left_anti") \
                                 .withColumn(op_col, F.lit("I"))
        insert_count = df_inserts.count()
        logging.info(f"[CDC] Inserts: {insert_count} rows ({time.time() - cdc_start:.1f}s)")
        
        # 6b. DELETES: target rows whose keys don't exist in source
        # For deletes, we need target column values (source is NULL)
        del_key_join = [target_with_hash[tk] == df_with_hash[sk] for sk, tk in zip(s_kc, t_kc)]
        del_key_cond = del_key_join[0]
        for c in del_key_join[1:]:
            del_key_cond = del_key_cond & c
        
        df_deletes_raw = target_with_hash.join(F.broadcast(df_with_hash.select(s_kc)), del_key_cond, "left_anti")
        # Project delete rows into source schema (use target values)
        del_cols = []
        for c in df.columns:
            if c in target_df.columns:
                del_cols.append(F.col(c))
            else:
                del_cols.append(F.lit(None).alias(c))
        df_deletes = df_deletes_raw.select(del_cols) \
                                   .withColumn("__src_hash", F.lit("")) \
                                   .withColumn(op_col, F.lit("D"))
        delete_count = df_deletes.count()
        logging.info(f"[CDC] Deletes: {delete_count} rows ({time.time() - cdc_start:.1f}s)")
        
        # 6c. UPDATES: matching keys where checksum differs
        # Inner join, then filter where hashes are different
        df_updates = df_with_hash.join(F.broadcast(target_with_hash.select(t_kc + ["__tgt_hash"])), key_cond, "inner") \
                                 .filter(F.col("__src_hash") != F.col("__tgt_hash")) \
                                 .drop("__tgt_hash") \
                                 .withColumn(op_col, F.lit("U"))
        update_count = df_updates.count()
        logging.info(f"[CDC] Updates: {update_count} rows ({time.time() - cdc_start:.1f}s)")
        
        # --- 7. Union only the changed rows ---
        # Ensure all DataFrames have the same columns before union
        all_cols = df.columns + ["__src_hash", op_col]
        
        result = df_inserts.select(all_cols) \
                    .unionByName(df_deletes.select(all_cols)) \
                    .unionByName(df_updates.select(all_cols))
        
        # Add debug columns (empty for performance, the hash is the checksum)
        result = result.withColumn("__debug_src", F.col("__src_hash")) \
                       .withColumn("__debug_tgt", F.lit("")) \
                       .drop("__src_hash")
        
        target_df.unpersist()
        
        total_changes = insert_count + delete_count + update_count
        logging.info(f"[CDC] Complete: {total_changes} changes (I={insert_count}, U={update_count}, D={delete_count}) "
                     f"out of {df.count()} source rows ({time.time() - cdc_start:.1f}s)")
        return result


class TransformationFactory:
    _handlers = {'column': ColumnTransformationHandler, 'row': RowTransformationHandler, 'lookup': LookupTransformationHandler, 'cdc': CDCTransformationHandler, 'cdc_compare': CDCTransformationHandler}
    @classmethod
    def get_handler(cls, t_type: str) -> TransformationHandler:
        if t_type not in cls._handlers: raise ValueError(f"Unsupported transform: {t_type}")
        return cls._handlers[t_type]()

def apply_all_transformations(engine, df: DataFrame, config: Dict, context: Dict = None) -> DataFrame:
    """Helper to apply array of transformations checking scope"""
    pipeline_type = config.get('pipeline', {}).get('pipeline_type')
    is_file_target = pipeline_type in ['FILE_TO_FILE', 'DB_TO_FILE']
    
    transforms = config.get('transformations', [])
    for t_conf in transforms:
        t_type = t_conf.get('type')
        if is_file_target and t_type in ['cdc', 'cdc_compare', 'lookup']:
            logging.warning(f"Skipping {t_type} transformation because Target is FILE. (CDC and lookups are out of scope for FILE targets).")
            continue
        handler = TransformationFactory.get_handler(t_type)
        df = handler.apply(df, t_conf, context)
        
    return df

def apply_field_mappings(df: DataFrame, config: Dict) -> DataFrame:
    mappings = config.get('field_mappings', [])
    if not mappings: return df
    s_map = sorted(mappings, key=lambda x: x.get('order', 0))
    selects = []
    selected_cols = set()
    for m in s_map:
        scol, tcol = m.get('source_column'), m.get('target_column', m.get('source_column'))
        if scol in df.columns and scol not in selected_cols:
            selects.append(F.col(scol).alias(tcol) if scol != tcol else F.col(scol))
            selected_cols.add(scol)
    for d in ['__debug_src', '__debug_tgt']:
        if d in df.columns and d not in selected_cols: 
            selects.append(F.col(d))
            selected_cols.add(d)
    return df.select(*selects)
