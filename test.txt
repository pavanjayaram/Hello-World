import csv
from datetime import datetime
import os

def generate_csv_report(test_name, mismatches):
    filename = f"reports/{test_name}_report.csv"
    os.makedirs("reports", exist_ok=True)

    with open(filename, "w", newline="") as file:
        writer = csv.writer(file)
        writer.writerow(["Field", "DEV", "UAT"])

        for m in mismatches:
            writer.writerow([m["field"], m["dev"], m["uat"]])

    return filename


def generate_html_report(test_name, mismatches, match_percent):
    filename = f"reports/{test_name}_report.html"

    html = f"""
    <html>
    <body>
    <h2>Test: {test_name}</h2>
    <h3>Match %: {match_percent:.2f}%</h3>
    <table border="1">
    <tr><th>Field</th><th>DEV</th><th>UAT</th></tr>
    """

    for m in mismatches:
        html += f"<tr><td>{m['field']}</td><td>{m['dev']}</td><td>{m['uat']}</td></tr>"

    html += "</table></body></html>"

    with open(filename, "w") as f:
        f.write(html)

    return filename
