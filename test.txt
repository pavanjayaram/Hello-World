def execute_test(test, config):
    test_name = test["test_name"]

    dev_payload = read_xml_file(test["source_request_file"])
    uat_payload = read_xml_file(test["target_request_file"])

    dev_res = call_api(config["dev_url"], dev_payload, config["soap_action"])
    uat_res = call_api(config["uat_url"], uat_payload, config["soap_action"])

    dev_dict = xml_to_dict(dev_res)
    uat_dict = xml_to_dict(uat_res)

    result = deep_compare(
        dev_dict,
        uat_dict,
        ignore_fields=config["ignore_fields"],
        strict=True
    )

    mismatches = result["mismatch"]
    matches = result["match"]

    total = len(matches) + len(mismatches)
    score = (len(matches) / total) * 100 if total else 0

    status = "Passed" if not mismatches else "Failed"

    csv_file, html_file = generate_reports(test_name, mismatches, matches, score)

    print(f"{test_name} → {status} ({score:.2f}%)")
