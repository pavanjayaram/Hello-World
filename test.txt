import os

def load_test_cases(base_path):
    test_cases = []

    for folder in os.listdir(base_path):
        folder_path = os.path.join(base_path, folder)

        if os.path.isdir(folder_path):
            source_file = os.path.join(folder_path, "source.xml")
            target_file = os.path.join(folder_path, "target.xml")

            if os.path.exists(source_file) and os.path.exists(target_file):
                test_cases.append({
                    "test_name": folder,
                    "source_request_file": source_file,
                    "target_request_file": target_file
                })

    return test_cases
