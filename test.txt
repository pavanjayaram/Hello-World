import csv
import os
from src.scoring import classify_severity

def generate_reports(test_name, mismatches, score):
    os.makedirs("reports", exist_ok=True)

    csv_file = f"reports/{test_name}.csv"
    html_file = f"reports/{test_name}.html"

    # CSV
    with open(csv_file, "w", newline="") as f:
        writer = csv.writer(f)
        writer.writerow(["Field", "DEV", "UAT", "Type", "Severity"])

        for m in mismatches:
            field = m.get("field")
            dev = m.get("dev")
            uat = m.get("uat")
            mtype = m.get("type")
            severity = classify_severity(field)

            writer.writerow([field, dev, uat, mtype, severity])

    # HTML
    html = f"<h2>{test_name}</h2><h3>Score: {score:.2f}%</h3><table border=1>"
    html += "<tr><th>Field</th><th>DEV</th><th>UAT</th><th>Type</th><th>Severity</th></tr>"

    for m in mismatches:
        field = m.get("field")
        dev = m.get("dev")
        uat = m.get("uat")
        mtype = m.get("type")
        severity = classify_severity(field)

        color = "red" if severity == "HIGH" else "orange"

        html += f"""
        <tr style='color:{color}'>
            <td>{field}</td>
            <td>{dev}</td>
            <td>{uat}</td>
            <td>{mtype}</td>
            <td>{severity}</td>
        </tr>
        """

    html += "</table>"

    with open(html_file, "w") as f:
        f.write(html)

    return csv_file, html_file
