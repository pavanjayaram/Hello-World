def check_api(cfg):
    try:
        name = cfg.get("name", "UNKNOWN")
        method = cfg.get("method", "GET").upper()

        env_config = cfg.get("environments", {}).get(ENV)

        if not env_config:
            return False, f"{name} -> No config for ENV {ENV}"

        url = env_config["url"]

        log(f"[{ENV}] Checking {name}...")

        response = requests.get(
            url,
            timeout=TIMEOUT,
            verify=False
        )

        if response.status_code == 200:
            return True, f"{name} -> OK (200)"
        else:
            return False, f"{name} -> FAILED ({response.status_code})"

    except Exception as e:
        return False, f"{name} -> ERROR ({str(e)})"
