#!/usr/bin/env bash
# =============================================================================
#  PySpark ETL Pipeline Runner — Shell Wrapper for Autosys
# =============================================================================
#
#  This script wraps run_pipeline.py so Autosys can invoke it directly.
#  It handles Python environment setup and forwards all arguments.
#
#  EXIT CODES (passed through from run_pipeline.py):
#    0 = Pipeline completed successfully
#    1 = Pipeline execution failed (runtime error)
#    2 = Bad arguments, missing files, or invalid configuration
#
# =============================================================================
#
#  Autosys JIL Example:
#  --------------------
#  insert_job: ETL_DAILY_LOAD
#  job_type: CMD
#  command: /opt/etl/standalone/run_pipeline.sh --config /opt/etl/configs/daily_load.json --params /opt/etl/parameterfile.txt --env-file /opt/etl/.env --log-dir /opt/etl/logs
#  machine: etl-server-01
#  owner: etl_user
#  date_conditions: 1
#  days_of_week: mo,tu,we,th,fr
#  start_times: "06:00"
#  std_out_file: /opt/etl/logs/autosys_stdout.log
#  std_err_file: /opt/etl/logs/autosys_stderr.log
#  alarm_if_fail: 1
#
# =============================================================================

set -euo pipefail

# Resolve the directory where this script lives
SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
PROJECT_ROOT="$(dirname "$SCRIPT_DIR")"
BACKEND_DIR="${PROJECT_ROOT}/backend"

echo "============================================"
echo "  PySpark ETL Standalone Runner"
echo "  Script Dir  : ${SCRIPT_DIR}"
echo "  Project Root: ${PROJECT_ROOT}"
echo "  Backend Dir : ${BACKEND_DIR}"
echo "  Timestamp   : $(date '+%Y-%m-%d %H:%M:%S')"
echo "============================================"

# ---------------------------------------------------------------------------
# 1. Set PYTHONPATH so Python can find backend modules
# ---------------------------------------------------------------------------
export PYTHONPATH="${BACKEND_DIR}:${PYTHONPATH:-}"

# ---------------------------------------------------------------------------
# 2. Activate virtual environment if it exists
# ---------------------------------------------------------------------------
VENV_ACTIVATE="${PROJECT_ROOT}/.venv/bin/activate"
if [ -f "$VENV_ACTIVATE" ]; then
    echo "[Setup] Activating virtual environment: ${VENV_ACTIVATE}"
    # shellcheck disable=SC1090
    source "$VENV_ACTIVATE"
else
    echo "[Setup] No virtual environment found at ${VENV_ACTIVATE}, using system Python"
fi

echo "[Setup] Python: $(which python3)"
echo "[Setup] Python version: $(python3 --version)"

# ---------------------------------------------------------------------------
# 3. Run the pipeline
# ---------------------------------------------------------------------------
echo "[Runner] Executing: spark-submit --master \"local[*]\" ${SCRIPT_DIR}/run_pipeline.py $*"
echo ""

spark-submit --master "local[*]" "${SCRIPT_DIR}/run_pipeline.py" "$@"
EXIT_CODE=$?

echo ""
echo "[Runner] Pipeline exited with code: ${EXIT_CODE}"
exit ${EXIT_CODE}
