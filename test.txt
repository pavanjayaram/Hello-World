def deep_compare(dev, uat, path="", ignore_fields=None, strict=True):
    if ignore_fields is None:
        ignore_fields = []

    result = {
        "mismatch": [],
        "match": []
    }

    if is_ignored(path, ignore_fields):
        return result

    if type(dev) != type(uat):
        result["mismatch"].append({
            "field": path,
            "dev": dev,
            "uat": uat,
            "type": "TYPE_MISMATCH"
        })
        return result

    if isinstance(dev, dict):
        all_keys = set(dev.keys()).union(set(uat.keys()))

        for key in all_keys:
            new_path = f"{path}.{key}" if path else key

            if key not in dev:
                result["mismatch"].append({
                    "field": new_path,
                    "dev": None,
                    "uat": uat[key],
                    "type": "MISSING_IN_DEV"
                })
            elif key not in uat:
                result["mismatch"].append({
                    "field": new_path,
                    "dev": dev[key],
                    "uat": None,
                    "type": "MISSING_IN_UAT"
                })
            else:
                sub = deep_compare(dev[key], uat[key], new_path, ignore_fields, strict)
                result["mismatch"].extend(sub["mismatch"])
                result["match"].extend(sub["match"])

    elif isinstance(dev, list):
        # keep your existing list logic (extend mismatch)
        pass

    else:
        leaf = compare_values(dev, uat, path, strict)
        result["mismatch"].extend(leaf["mismatch"])
        result["match"].extend(leaf["match"])

    return result
